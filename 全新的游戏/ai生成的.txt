// src/systems/inventory/ItemManager.ts
import { GameSession, InventoryGrid, ItemInstance, addItemToGrid, removeItemFromGrid, moveItemBetweenGrids } from '../../core/Session';
import itemsConfig from '../../config/items.json';

export interface UseItemResult {
  success: boolean;
  message?: string;
  healAmount?: number;
  ammoAmount?: number;
}

export class ItemManager {
  constructor(private session: GameSession) {}

  // ---- 增删改（直接操作 Session.inventories） ----
  
  /** 添加物品到指定网格（自动堆叠） */
  addItem(layer: keyof GameSession['inventories'], itemId: string, count: number): boolean {
    const grid = this.session.inventories[layer];
    if (!grid) return false;
    const maxStack = itemsConfig.items.find(i => i.id === itemId)?.maxStack || 99;
    return addItemToGrid(grid as InventoryGrid, itemId, count, maxStack);
  }

  /** 从指定网格移除物品 */
  removeItem(layer: keyof GameSession['inventories'], itemId: string, count: number): boolean {
    const grid = this.session.inventories[layer];
    if (!grid) return false;
    return removeItemFromGrid(grid as InventoryGrid, itemId, count);
  }

  /** 移动物品（L1↔L2↔L3↔L4，跨层） */
  moveItem(srcLayer: keyof GameSession['inventories'], dstLayer: keyof GameSession['inventories'], itemId: string, count: number): boolean {
    const srcGrid = this.session.inventories[srcLayer];
    const dstGrid = this.session.inventories[dstLayer];
    if (!srcGrid || !dstGrid) return false;
    const maxStack = itemsConfig.items.find(i => i.id === itemId)?.maxStack || 99;
    return moveItemBetweenGrids(srcGrid as InventoryGrid, dstGrid as InventoryGrid, itemId, count, maxStack);
  }

  /** 获取网格中所有物品列表（供 UI 渲染） */
  getItems(layer: keyof GameSession['inventories']): { itemId: string; stackSize: number; row: number; col: number }[] {
    const grid = this.session.inventories[layer];
    const result: { itemId: string; stackSize: number; row: number; col: number }[] = [];
    if (!grid) return result;
    for (let r = 0; r < grid.length; r++) {
      for (let c = 0; c < grid[r].length; c++) {
        const slot = grid[r][c];
        if (slot) result.push({ itemId: slot.itemId, stackSize: slot.stackSize, row: r, col: c });
      }
    }
    return result;
  }

  // ---- 物品使用（消耗品/材料） ----
  
  /** 使用物品（由 UI 层调用） */
  useItem(layer: keyof GameSession['inventories'], row: number, col: number): UseItemResult {
    const grid = this.session.inventories[layer];
    if (!grid || !grid[row] || !grid[row][col]) {
      return { success: false, message: '物品不存在' };
    }
    const slot = grid[row][col]!;
    const config = itemsConfig.items.find(i => i.id === slot.itemId);
    if (!config || config.type !== 'consumable') {
      return { success: false, message: '该物品无法使用' };
    }

    // ★ 根据配置执行效果
    let result: UseItemResult = { success: true };
    if (config.effect?.heal) {
      const healed = Math.min(this.session.player.maxHp - this.session.player.hp, config.effect.heal);
      this.session.player.hp += healed;
      result.healAmount = healed;
    }
    if (config.effect?.ammo) {
      // 弹药逻辑后续挂接玩家武器系统
      result.ammoAmount = config.effect.ammo;
    }

    // 扣减物品
    this.removeItem(layer, slot.itemId, 1);
    return result;
  }

  /** 检查是否有足够空间（常用于拾取前判断） */
  hasSpace(layer: keyof GameSession['inventories'], itemId: string, count: number): boolean {
    // 简单实现：遍历找空位
    const grid = this.session.inventories[layer];
    if (!grid) return false;
    let emptySlots = 0;
    for (const row of grid) {
      for (const cell of row) {
        if (cell === null) emptySlots++;
        else if (cell.itemId === itemId) {
          const maxStack = itemsConfig.items.find(i => i.id === itemId)?.maxStack || 99;
          emptySlots += maxStack - cell.stackSize;
        }
        if (emptySlots >= count) return true;
      }
    }
    return false;
  }
}
2. CraftingManager（合成管理）

// src/systems/inventory/CraftingManager.ts
import { GameSession } from '../../core/Session';
import { ItemManager } from './ItemManager';
import craftingRecipes from '../../config/craftRecipes.json';

export interface Recipe {
  id: string;
  name: string;
  inputs: { itemId: string; count: number }[];
  output: { itemId: string; count: number };
  station: 'ship' | 'portable'; // 舰船合成台 / 便携合成台
}

export class CraftingManager {
  private recipes: Recipe[] = craftingRecipes.recipes;

  constructor(private session: GameSession, private itemManager: ItemManager) {}

  /** 获取当前合成台可用的配方 */
  getAvailableRecipes(station: 'ship' | 'portable'): Recipe[] {
    return this.recipes.filter(r => r.station === station || r.station === 'ship');
  }

  /** 检查玩家是否拥有足够材料 */
  canCraft(recipeId: string, srcLayer: keyof GameSession['inventories']): boolean {
    const recipe = this.recipes.find(r => r.id === recipeId);
    if (!recipe) return false;
    const grid = this.session.inventories[srcLayer];
    if (!grid) return false;

    // 临时拷贝网格计数
    const available = new Map<string, number>();
    for (const row of grid) {
      for (const cell of row) {
        if (cell) available.set(cell.itemId, (available.get(cell.itemId) || 0) + cell.stackSize);
      }
    }
    for (const input of recipe.inputs) {
      if ((available.get(input.itemId) || 0) < input.count) return false;
    }
    return true;
  }

  /** 执行合成 */
  craft(recipeId: string, srcLayer: keyof GameSession['inventories'], dstLayer: keyof GameSession['inventories']): boolean {
    const recipe = this.recipes.find(r => r.id === recipeId);
    if (!recipe || !this.canCraft(recipeId, srcLayer)) return false;

    // 扣材料
    for (const input of recipe.inputs) {
      this.itemManager.removeItem(srcLayer, input.itemId, input.count);
    }

    // 产出物（先判断目标网格是否有空间）
    const output = recipe.output;
    if (!this.itemManager.hasSpace(dstLayer, output.itemId, output.count)) {
      // 回滚材料
      for (const input of recipe.inputs) {
        this.itemManager.addItem(srcLayer, input.itemId, input.count);
      }
      return false;
    }
    this.itemManager.addItem(dstLayer, output.itemId, output.count);
    return true;
  }
}

// src/systems/interaction/InteractionManager.ts
import { GameSession } from '../../core/Session';
import { ItemManager } from '../inventory/ItemManager';

export interface Interactable {
  id: string;
  type: 'item' | 'npc' | 'crafting_table' | 'container' | 'exit';
  data?: any;
}

export interface InteractionContext {
  session: GameSession;
  itemManager: ItemManager;
  // 对话系统后续接入
}

export class InteractionManager {
  constructor(private ctx: InteractionContext) {}

  // ---- 核心：处理任何交互 ----
  interact(target: Interactable): { action: string; payload?: any; message?: string } {
    switch (target.type) {
      case 'item':
        return this.handleItem(target);
      case 'crafting_table':
        return this.handleCraftingTable(target);
      case 'npc':
        return this.handleNpc(target);
      case 'container':
        return this.handleContainer(target);
      case 'exit':
        return { action: 'exit', message: '返回舰船' };
      default:
        return { action: 'none', message: '无法交互' };
    }
  }

  private handleItem(target: Interactable): { action: string; payload?: any } {
    // 物品交互：返回物品详情，UI 层决定展示什么
    return {
      action: 'show_item_detail',
      payload: {
        itemId: target.data?.itemId,
        grid: target.data?.grid,
        row: target.data?.row,
        col: target.data?.col,
      }
    };
  }

  private handleCraftingTable(target: Interactable): { action: string; payload?: any } {
    return {
      action: 'open_crafting',
      payload: {
        station: target.data?.level || 'portable', // 'ship' 或 'portable'
      }
    };
  }

  private handleNpc(target: Interactable): { action: string; payload?: any } {
    return {
      action: 'start_dialogue',
      payload: {
        npcId: target.data?.npcId,
        dialogueTree: target.data?.dialogueTree || 'default',
      }
    };
  }

  private handleContainer(target: Interactable): { action: string; payload?: any } {
    // 容器（箱子/补给箱）交互
    return {
      action: 'open_container',
      payload: {
        containerId: target.data?.containerId,
        inventory: target.data?.inventory, // 容器内的物品列表
      }
    };
  }

  // ---- 快捷辅助：拾取（WorldMode 专用） ----
  pickup(itemId: string, fromWorld: boolean = true): boolean {
    // 从世界拾取 → 进入玩家背包 (L3)
    return this.ctx.itemManager.addItem('player', itemId, 1);
  }
}

import { BaseInteractionUI } from '../BaseInteractionUI';
import { Minimap } from './Minimap';
import { PlayerHud } from './PlayerHud';
import { Crosshair } from './Crosshair';
import { FloatingText } from './FloatingText';
import { GameSession } from '../../core/Session';
import { ItemManager } from '../../systems/inventory/ItemManager';

export class WorldUIManager extends BaseInteractionUI {

  private hud: PlayerHud;
  private crosshair: Crosshair;
  private floatingTexts: FloatingText[] = [];

  // ★ 交互提示（按E拾取）
  private interactPrompt: HTMLDivElement;

  constructor(
    private session: GameSession,
    private itemManager: ItemManager,
    private interactionManager: InteractionManager,
  ) {
    super();
    this.minimap = new Minimap(/* raster */);
    this.hud = new PlayerHud();
    this.crosshair = new Crosshair();
    this.buildInteractPrompt();
  }

  private buildInteractPrompt(): void {
    this.interactPrompt = document.createElement('div');
    this.interactPrompt.style.cssText = `
      position: fixed; bottom: 120px; left: 50%; transform: translateX(-50%);
      color: #fff; background: rgba(0,0,0,0.7); padding: 8px 16px;
      border-radius: 4px; display: none; z-index: 200;
      font-size: 14px; pointer-events: none;
    `;
    this.interactPrompt.textContent = '按 E 拾取';
    document.body.appendChild(this.interactPrompt);
  }

  // ---- 每帧更新（高频） ----
  update(dt: number, ctx: {
    px: number; pz: number; yaw: number;
    entities: any[];
    hp: number; maxHp: number;
    nearbyItem?: { itemId: string; distance: number };
  }): void {
    this.minimap.update(ctx.px, ctx.pz, ctx.yaw, ctx.entities);
    this.hud.update(ctx.hp, ctx.maxHp);

    // 交互提示
    if (ctx.nearbyItem && ctx.nearbyItem.distance < 2) {
      this.interactPrompt.textContent = `按 E 拾取 ${ctx.nearbyItem.itemId}`;
      this.interactPrompt.style.display = 'block';
    } else {
      this.interactPrompt.style.display = 'none';
    }

    // 浮动文字（自动飘动消失）
    for (let i = this.floatingTexts.length - 1; i >= 0; i--) {
      if (this.floatingTexts[i].update(dt)) {
        this.floatingTexts[i].dispose();
        this.floatingTexts.splice(i, 1);
      }
    }
  }

  // ---- ★ 显示浮动伤害数字（WorldMode 调用） ----
  showFloatingText(x: number, y: number, text: string, color: string = '#ff4444'): void {
    const ft = new FloatingText(x, y, text, color);
    this.floatingTexts.push(ft);
  }

  // ---- ★ 打开对话（WorldMode 触发，轻量版，不阻塞操作） ----
  openDialogue(npcId: string, text: string): void {
    // 世界中的对话不阻塞操作，显示在角落
    const bubble = document.createElement('div');
    bubble.style.cssText = `
      position: fixed; bottom: 200px; left: 50%; transform: translateX(-50%);
      background: rgba(0,0,0,0.85); color: #fff; padding: 16px 24px;
      border-radius: 8px; max-width: 400px; z-index: 150;
      border: 1px solid #4466aa;
    `;
    bubble.textContent = `${npcId}: ${text}`;
    document.body.appendChild(bubble);
    setTimeout(() => bubble.remove(), 3000);
  }

  // ---- ★ 打开容器/拾取结果（WorldMode 触发） ----
  showPickupResult(itemId: string, success: boolean): void {
    const msg = success ? `拾取了 ${itemId}` : `背包已满，无法拾取 ${itemId}`;
    // 用浮动文字展示
    this.showFloatingText(0, -50, msg, success ? '#44dd88' : '#ff4444');
  }

  override dispose(): void {
    super.dispose();
    this.minimap.dispose();
    this.hud.dispose();
    this.crosshair.dispose();
    this.interactPrompt.remove();
    this.floatingTexts.forEach(ft => ft.dispose());
    this.floatingTexts = [];
  }
}

// src/modes/ShipMode.ts（片段）
export class ShipMode implements IGameMode {
  private itemManager!: ItemManager;
  private craftingManager!: CraftingManager;
  private interactionManager!: InteractionManager;
  private uiManager!: ShipUIManager;

  enter(ctx: IGameModeContext): void {
    // ① 初始化业务逻辑层

    this.craftingManager = new CraftingManager(ctx.session, this.itemManager);
    this.interactionManager = new InteractionManager({

      itemManager: this.itemManager,
    });

    // ② 初始化 UI 层
    this.uiManager = new ShipUIManager(
      ctx.session,
      this.itemManager,
      this.craftingManager,
      this.interactionManager,
    );

    // ③ 设置 3D 点击交互：点击建筑 → 调用 interactionManager.interact()
    this.setupRaycaster((target) => {
      const result = this.interactionManager.interact(target);
      // 根据 result.action 决定 UI 行为
      if (result.action === 'open_crafting') {
        this.uiManager.openCrafting(result.payload.station);
      } else if (result.action === 'show_item_detail') {
        this.uiManager.openItemDetail(
          result.payload.grid,
          result.payload.row,
          result.payload.col,
        );
      } else if (result.action === 'start_dialogue') {
        this.uiManager.openDialogue(result.payload.npcId, result.payload.dialogueTree);
      }
    });
  }

  exit(): void {
    this.uiManager.dispose();
    // 业务逻辑层无需销毁（无状态）
  }
}

// src/modes/WorldMode.ts（片段）
export class WorldMode implements IGameMode {
  private itemManager!: ItemManager;
  private craftingManager!: CraftingManager;
  private interactionManager!: InteractionManager;
  private uiManager!: WorldUIManager;

  enter(ctx: WorldModeEnterContext): void {
    // ① 业务逻辑层（与 ShipMode 共享同样的类，但实例独立）
    this.itemManager = new ItemManager(ctx.session);
    this.craftingManager = new CraftingManager(ctx.session, this.itemManager);
    this.interactionManager = new InteractionManager({
      session: ctx.session,
      itemManager: this.itemManager,
    });

    // ② UI 层（世界专用）
    this.uiManager = new WorldUIManager(ctx.session, this.itemManager, this.interactionManager);
  }

  update(dt: number): void {
    // ... 现有战斗逻辑 ...

    // ★ 检测附近可交互物品
    const nearbyItem = this.findNearestItem();
    this.uiManager.update(dt, {
      px: this.player.position.x,
      pz: this.player.position.z,
      yaw: this.cameraCtrl.worldYaw,
      entities: this.entities.allBases(),
      hp: this.player.hp,
      maxHp: this.player.maxHp,
      nearbyItem: nearbyItem,
    });

    // ★ 按 E 交互
    if (input.held.interact && nearbyItem) {
      const result = this.interactionManager.interact({
        type: 'item',
        id: nearbyItem.itemId!,
        data: { itemId: nearbyItem.itemId!, grid: 'player' },
      });
      if (result.action === 'show_item_detail') {
        // 世界模式中拾取直接入包，不弹详情
        const success = this.itemManager.addItem('player', nearbyItem.itemId!, 1);
        this.uiManager.showPickupResult(nearbyItem.itemId!, success);
        if (success) nearbyItem.dispose(); // 销毁世界中的物品实体
      }
    }
  }

  exit(): void {
    this.uiManager.dispose();
    // 业务逻辑层随模式销毁
  }
}

✅ 这套架构的核心优势
优势	说明
业务逻辑与 UI 完全解耦	ItemManager/CraftingManager 不知道自己在舰船还是世界中被调用。
舰船与世界 UI 独立进化	ShipUIManager 可以添加复杂的拖拽、右键菜单；WorldUIManager 只需轻量 HUD。
共用的 UI 基类	BaseInteractionUI 统一管理弹窗栈和输入阻塞，新面板只需继承。
共用的 UI 组件	renderItemSlot 在舰船和世界的背包/拾取提示中保持一致。
开闭原则	加新交互（如“商店”）只需：① 在 Interactable 加类型；② 在 InteractionManager 加 handler；③ 在 UI 加面板渲染。
可测试	ItemManager 可以脱离 UI 单独测试，只需传入 Mock Session。
你现在可以并行开发：一个人（或同一时间）做 ShipUIManager 的复杂面板，另一个人做 WorldUIManager 的 HUD，底层 ItemManager 共用，互不干扰。
