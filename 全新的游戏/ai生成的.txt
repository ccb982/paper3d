我们来把这四层背包体系彻底理清，并定义它们的数据结构和交互规则。

🎒 四层背包定义与职责
层级	名称	容量	数据生命周期	可访问场景
L1	舰船基地仓库	超大（几乎无限）	永久存档（跨天、跨局）	仅在 ShipMode（舰船） 可访问
L2	探索飞船仓库	适中（如 8×10 格）	永久存档（跨天）	ShipMode + WorldMode（地图上）
L3	玩家背包	较小（如 4×6 格）	永久存档（跨天）	ShipMode + WorldMode
L4	队友背包	各不相同（由干员配置决定）	永久存档（跨天）	ShipMode + WorldMode
🔄 数据流转与存取规则（关键）
1. 出战（舰船 → 大世界）

玩家从 L1（基地） 挑选物资，转移到 L2（飞船仓库） 和 L3（玩家背包）。

L1 不进入地图。地图上只加载 Session 中的 L2、L3、L4。

地图种子由 day 决定，与背包无关。

2. 回归（大世界 → 舰船）

玩家回到舰船时，L2、L3、L4 中所有物品保留（不自动上缴）。

玩家可以在舰船上手动把 L2/L3/L4 的物品转移到 L1（仓库）进行囤积。

存档触发点：从 WorldMode 回到 ShipMode 的那一刻，全量保存 L1~L4 的所有数据。

3. 队友背包（L4）的特殊性

每个友军干员配置里有一个 backpackSlots 字段（如 6 格）。

如果该干员未被部署（在舰船待命），其背包依然存在，可在舰船界面整理。

如果该干员被部署到地图上（AI 跟随），其背包自动合并为队伍可用容量的一部分（玩家可以放入/取出物品，相当于随身携带的额外仓库）。

如果干员阵亡/离队，其背包中的物品需要掉落或返还（规则待定，先设计成“返还到飞船仓库”）。

/** 创建空网格 */
export function createEmptyGrid(rows: number, cols: number): InventoryGrid {
  return Array.from({ length: rows }, () => Array(cols).fill(null));
}

/** 深拷贝网格（防御性） */
export function cloneGrid(grid: InventoryGrid): InventoryGrid {
  return grid.map((row) => row.map((slot) => (slot ? { ...slot } : null)));
}

/** 统计网格中的物品总数 */
export function countItemsInGrid(grid: InventoryGrid): number {
  let count = 0;
  for (const row of grid) {
    for (const slot of row) {
      if (slot !== null) count++;
    }
  }
  return count;
}

/** 在网格中查找特定物品的第一个位置 */
export function findItemInGrid(
  grid: InventoryGrid,
  itemId: string,
): { row: number; col: number; item: ItemInstance } | null {
  for (let r = 0; r < grid.length; r++) {
    for (let c = 0; c < grid[r].length; c++) {
      const slot = grid[r][c];
      if (slot && slot.itemId === itemId) {
        return { row: r, col: c, item: slot };
      }
    }
  }
  return null;
}

/** 找第一个空位 */
export function findEmptySlot(grid: InventoryGrid): { row: number; col: number } | null {
  for (let r = 0; r < grid.length; r++) {
    for (let c = 0; c < grid[r].length; c++) {
      if (grid[r][c] === null) {
        return { row: r, col: c };
      }
    }
  }
  return null;
}

/** 新增物品到网格（自动堆叠，否则放空位） */
export function addItemToGrid(
  grid: InventoryGrid,
  itemId: string,
  stackSize: number,
  maxStack: number = 99,
): boolean {
  // 1. 先堆叠到已有同物品堆
  for (let r = 0; r < grid.length; r++) {
    for (let c = 0; c < grid[r].length; c++) {
      const slot = grid[r][c];
      if (slot && slot.itemId === itemId && slot.stackSize < maxStack) {
        const space = maxStack - slot.stackSize;
        if (space >= stackSize) {
          slot.stackSize += stackSize;
          return true;
        } else {
          slot.stackSize = maxStack;
          stackSize -= space;
        }
      }
    }
  }

  // 2. 放空位
  while (stackSize > 0) {
    const pos = findEmptySlot(grid);
    if (!pos) return false; // 背包满了
    const put = Math.min(stackSize, maxStack);
    grid[pos.row][pos.col] = { itemId, stackSize: put };
    stackSize -= put;
  }
  return true;
}

/** 战斗属性 = 基础属性 + 所有藏品加成（临时，不存档） */
export interface PlayerCombatStats {
  hp: number;
  maxHp: number;
  attackPower: number;
  defense: number;
}

/** 藏品配置条目（由外部配置表提供） */
export interface RelicConfigEntry {
  id: string;
  name: string;
  type: 'carry' | 'permanent';
  description: string;
  effect?: {
    // 携带型（carry）：固定数值
    attackBonus?: number;
    defenseBonus?: number;
    hpBonus?: number;
    // 永久型（permanent）：每日倍率
    multiplier?: number;
    // 永久型也可以有固定加成
    flatBonus?: {
      attackBonus?: number;
      defenseBonus?: number;
      hpBonus?: number;
    };
  };
}

/**
 * ★ 从 Session 计算战斗属性（出击时调用）
 * 完全配置驱动，没有任何硬编码的特例
 */
export function computeCombatStats(
  session: GameSession,
  relicConfig: Record<string, RelicConfigEntry>,
): PlayerCombatStats {
  const base = session.player;
  const owned = session.relics.owned;
  const day = session.meta.day;

  let bonusAttack = 0;
  let bonusDefense = 0;
  let bonusMaxHp = 0;
  let multiplier = 1;

  for (const id of owned) {
    const cfg = relicConfig[id];
    if (!cfg) continue;

    if (cfg.type === 'carry') {
      // 携带型：固定数值加成
      bonusAttack += cfg.effect?.attackBonus || 0;
      bonusDefense += cfg.effect?.defenseBonus || 0;
      bonusMaxHp += cfg.effect?.hpBonus || 0;
    } else if (cfg.type === 'permanent') {
      // 永久型：每日倍率（如黑冠）
      if (cfg.effect?.multiplier) {
        multiplier *= Math.pow(cfg.effect.multiplier, day);
      }
      // 永久型也可以有固定加成
      bonusAttack += cfg.effect?.flatBonus?.attackBonus || 0;
      bonusDefense += cfg.effect?.flatBonus?.defenseBonus || 0;
      bonusMaxHp += cfg.effect?.flatBonus?.hpBonus || 0;
    }
  }

  return {
    hp: base.hp,
    maxHp: Math.floor(base.maxHp * multiplier) + bonusMaxHp,
    attackPower: Math.floor(base.attackPower * multiplier) + bonusAttack,
    defense: Math.floor(base.defense * multiplier) + bonusDefense,
  };
}

// src/core/SaveSystem.ts
// ============================================================
// 存档系统 —— 唯一存档入口
// 触发时机：① 返回舰船 ② 抽卡后
// 存储介质：localStorage（Web）/ 后续可替换为 Adapter
// ============================================================

import type { GameSession } from './Session';

const STORAGE_KEY = 'arknights_rogue_save';

export const SaveSystem = {
  /**
   * ★ 保存存档（仅在安全点调用）
   * 调用前需确保 session 中不含任何运行时引用（Three/Rapier 对象）
   */
  save(session: GameSession): void {
    try {
      // 更新时间戳
      session.meta.lastSavedAt = new Date().toISOString();

      const json = JSON.stringify(session);
      localStorage.setItem(STORAGE_KEY, json);

      console.log(`[存档] ✅ 第 ${session.meta.day} 天已保存至舰船`);
      console.log(`[存档] 📦 背包物品: 舰船 ${countItems(session.inventories.base)} 件, ` +
        `飞船 ${countItems(session.inventories.ship)} 件, ` +
        `玩家 ${countItems(session.inventories.player)} 件`);
    } catch (e) {
      console.error('[存档] ❌ 保存失败:', e);
      // 可考虑降级方案（如内存缓存）
    }
  },

  /**
   * 读取存档
   * @returns 存档对象，不存在或损坏返回 null
   */
  load(): GameSession | null {
    try {
      const raw = localStorage.getItem(STORAGE_KEY);
      if (!raw) {
        console.log('[存档] 📭 未找到存档，将开始新游戏');
        return null;
      }

      const data = JSON.parse(raw) as GameSession;

      // ★ 版本兼容性检查（未来扩展）
      if (!data.meta || !data.meta.version) {
        console.warn('[存档] ⚠️ 存档版本不兼容，将丢弃');
        return null;
      }

      console.log(`[存档] 📖 读取成功，第 ${data.meta.day} 天`);
      return data;
    } catch (e) {
      console.error('[存档] ❌ 读取失败:', e);
      return null;
    }
  },

  /**
   * ★ 清除存档（舰船被毁 / 通关 / 手动开新局）
   */
  clear(): void {
    try {
      localStorage.removeItem(STORAGE_KEY);
      console.log('[存档] 🗑️ 存档已清除');
    } catch (e) {
      console.warn('[存档] 清除失败:', e);
    }
  },

  /**
   * 检查是否存在存档
   */
  hasSave(): boolean {
    return localStorage.getItem(STORAGE_KEY) !== null;
  },

  /**
   * 获取存档大小（调试用）
   */
  getSize(): number {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? raw.length : 0;
  },
};

import { SaveSystem } from './core/SaveSystem';
import { createNewSession, computeCombatStats, type GameSession } from './core/Session';
import { RELIC_CONFIG } from './config/relics';

async function boot() {
  // 1. 加载或创建存档
  let session = SaveSystem.load();
  if (!session) {
    session = createNewSession();
    SaveSystem.save(session); // ★ 首次保存
  }

  // 2. ★ 出击时计算战斗属性（只读，不修改 session）
  const combatStats = computeCombatStats(session, RELIC_CONFIG);
  console.log(`[出击] 最终属性: HP ${combatStats.maxHp}, 攻击 ${combatStats.attackPower}, 防御 ${combatStats.defense}`);

  // 3. 创建 WorldMode，传入 combatStats
  const mode = new WorldMode(
    scene, camera, renderer,
    // ...
    session.meta.day,      // 地图种子
    combatStats,           // 战斗属性
  );

  // 4. 返回舰船时保存
  function returnToShip() {
    // 从 WorldMode 读取最新状态
    session.player.hp = mode.getPlayerHp();
    // ★ 唯一存档点
    SaveSystem.save(session);
    // 切换回 ShipMode
  }
}

    content.innerHTML = `

        <div style="color: #ffaa44; font-weight: bold; font-size: 15px;">💠 藏品 (${this.session.relics.owned.length})</div>

      </div>

  setActionState(state: ActionFlowState): void {
    this.actionState = state;
    this.renderActionPanel();
  }

  // ---- 辅助函数 ----

  private countItems(grid: (any | null)[][]): number {
    let count = 0;
    for (const row of grid) {
      for (const cell of row) {
        if (cell !== null) count++;
      }
    }
    return count;
  }

  private listItems(grid: (any | null)[][]): string {
    const items: string[] = [];
    for (const row of grid) {
      for (const cell of row) {
        if (cell !== null) {
          items.push(`${cell.itemId} ×${cell.stackSize}`);
        }
      }
    }
    return items.length > 0 ? items.join('、') : '（空）';
  }

  // ---- 更新（外部驱动） ----

  update(session: GameSession): void {
    this.session = session;
    this.render();
  }

  // ---- 清理 ----

  dispose(): void {
    this.root.remove();
  }
}

      if (!this.session.relics.owned.includes(result.id)) {
        this.session.relics.owned.push(result.id);
      }

    this.session.gacha.pityCounter += 1;
    this.session.gacha.totalPulls += 1;

    SaveSystem.save(this.session);

    // 刷新 UI
    this.uiLayer.update(this.session);
    this.uiLayer.render();

    // 检查是否可出击
    if (this.session.allies.roster.length > 0) {
      this.uiLayer.setActionState('gacha_done');
    }

    }

  // ---- 交互 ----

  private setupInteraction(): void {
    const el = this.renderer.domElement;
    el.addEventListener('click', this.onClick.bind(this));
    el.addEventListener('pointermove', this.onPointerMove.bind(this));
  }

  private onClick(e: MouseEvent): void {
    const rect = this.renderer.domElement.getBoundingClientRect();
    this.pointer.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
    this.pointer.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;

    this.raycaster.setFromCamera(this.pointer, this.camera);

    // 检测所有可交互对象
    const targets: THREE.Object3D[] = [this.floor];
    for (const [, mesh] of this.facilities) {
      targets.push(mesh);
    }
    for (const npc of this.npcs) {
      targets.push(npc);
    }
    // 舰船主体也可以交互
    targets.push(this.shipModel);

    const intersects = this.raycaster.intersectObjects(targets, true);

    if (intersects.length > 0) {
      let obj = intersects[0].object;
      // 向上查找 userData
      while (obj && !obj.userData?.type) {
        obj = obj.parent as THREE.Object3D;
      }

      if (obj?.userData?.type) {
        const data = obj.userData;
        const point = intersects[0].point;

        if (data.type === 'facility') {
          this.onInteraction({
            type: 'click',
            target: 'facility',
            facilityType: data.facilityType,
            position: point,
            data: data,
          });
        } else if (data.type === 'npc') {
          this.onInteraction({
            type: 'click',
            target: 'npc',
            position: point,
            data: data,
          });
        } else if (data.type === 'floor') {
          this.onInteraction({
            type: 'click',
            target: 'floor',
            position: point,
          });
        }
      }
    }
  }

  private onPointerMove(e: MouseEvent): void {
    // Hover 检测（改变光标）
    const rect = this.renderer.domElement.getBoundingClientRect();
    this.pointer.x = ((e.clientX - rect.left) / rect.width) * 2 - 1;
    this.pointer.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;

    this.raycaster.setFromCamera(this.pointer, this.camera);

    const targets: THREE.Object3D[] = [this.floor];
    for (const [, mesh] of this.facilities) {
      targets.push(mesh);
    }
    for (const npc of this.npcs) {
      targets.push(npc);
    }

    const intersects = this.raycaster.intersectObjects(targets, true);
    const hit = intersects.length > 0;

    this.renderer.domElement.style.cursor = hit ? 'pointer' : 'default';

    // Hover 高亮效果（可扩展）
  }

  // ---- 更新 ----

  update(dt: number): void {
    const time = Date.now() / 1000;

    // 信号灯闪烁
    const signalLight = this.shipModel.children.find(
      (c) => c.userData?.isSignalLight,
    ) as THREE.Mesh;
    if (signalLight) {
      const intensity = 0.3 + Math.sin(time * 2) * 0.3;
      (signalLight.material as THREE.MeshStandardMaterial).emissiveIntensity = Math.max(0, intensity);
    }

    // NPC 浮动
    for (let i = 0; i < this.npcs.length; i++) {
      const npc = this.npcs[i];
      npc.position.y = Math.sin(time * 1.2 + i * 2) * 0.05;
    }

    // L1 仓库脉冲光
    const warehouse = this.facilities.get('warehouse');
    if (warehouse) {
      // 找到发光圆点
      const logo = warehouse.children.find(
        (c) => c.type === 'Mesh' && (c as THREE.Mesh).geometry.type === 'SphereGeometry',
      ) as THREE.Mesh;
      if (logo) {
        const intensity = 0.5 + Math.sin(time * 1.5) * 0.3;
        (logo.material as THREE.MeshStandardMaterial).emissiveIntensity = Math.max(0.1, intensity);
      }
    }
  }

  // ---- 清理 ----

  dispose(): void {
    const el = this.renderer.domElement;
    el.removeEventListener('click', this.onClick.bind(this));
    el.removeEventListener('pointermove', this.onPointerMove.bind(this));

    // 递归清理场景（简化）
    while (this.scene.children.length > 0) {
      const child = this.scene.children[0];
      this.scene.remove(child);
      if ((child as THREE.Mesh).geometry) {
        (child as THREE.Mesh).geometry?.dispose();
        if (Array.isArray((child as THREE.Mesh).material)) {
          (child as THREE.Mesh).material.forEach(m => m.dispose());
        } else {
          (child as THREE.Mesh).material?.dispose();
        }
      }
    }
  }
}

✅ 架构总结
层级	职责	关键交互
MapLayer	3D 场景渲染 + 点击检测	L1 仓库显眼建筑 + 出口可点击
UILayer	三大面板（行动/编队/干员）	显示数据、抽卡、触发出击
ShipMode	持有两层 + 交互总线	地图点击 → 打开仓库弹窗 / 切换面板
L1 仓库：在地图上是一个发光的半透明科技建筑，点击后弹出仓库界面，展示 L1 的巨量物品。
出口：在地图上是发光门 + 光柱，点击后切换到“行动”面板，引导玩家走完抽卡→出击流程。
