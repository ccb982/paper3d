export interface ExchangeEvent {
  /** 本拍发生的事件（UI/日志用）。★ **不含**"状态迁移"以外的任何控制指令 */
  kind: 'none' | 'state_change' | 'touchdown' | 'liftoff' | 'safe';
  side?: Side;
  note: string;
}

// ══════════════════════════════════════════════════════════════════
// ★★★ **验收表（VERIFY）—— 每个状态「什么算通过」的唯一真源**
// ══════════════════════════════════════════════════════════════════
//
//  迁移**只**由这张表驱动（用户 2026-10-06：「验收过了进下一个状态就行了」）。
//  **没有任何计时器推进状态**；`stateT` 只用于 ① 防抖（OSL `min_time_in_state`）
//  ② `Tmax` 超时回退（Vughuma 2022）。
//
//  判据类型全部来自文献（详见 `架构_v2_三模块协作.md` §3.4）：
//   · 接触事件 / 接地标志 ................ SCONE、EPFL、OSL、Vughuma 全都用
//   · 足底载荷阈值（体重归一）............ OSL 0.25/0.15/0.40 BW；SCONE stance/swing threshold
//   · 足相对身体的**矢状位置**（腿长归一）.. SCONE `sagittal_pos` 全部迁移判据
//   · 关节角阈值 + **角速度**阈值 ......... OSL `knee>50° ∧ knee_vel<3`；EPFL `LP` 用 −1°/s
//   · 离地净空 .......................... Saunders 1953 MFC = 5cm
//   ⚠ **没有任何一项用重心位置**（旧的 `comOverFootX/Z` 已退役：它会自己推着自己走，
//     实测永不收敛 —— 243mm，见 `架构_v2` §3.5 的审计）。
//   ⚠ MoS / 躯干倾角是**本 rig 特有的加强项**（四篇实现测不到这两个量）。
export interface VerifySpec {
  /** 项名（进 `rs.violations[].item`，UI 直接显示） */
  item: string;
  /** 该项怎么算通过。`ctx` 里是本拍算好的全部原始量。 */
  ok: (ctx: VerifyCtx) => boolean;
  /** 回读值（用于 `violations[].value`） */
  val: (ctx: VerifyCtx) => number;
  /** 门限（用于 `violations[].tol`） */
  tol: (ctx: VerifyCtx) => number;
  /** true = `hard` 项：连续越界超 `graceSec` ⇒ 进安全态 */
  hard?: boolean;
}

export interface VerifyCtx {
  rs: RigState;
  cfg: GaitConfig;
  state: GaitState;
  /** 承重腿（载荷优势腿） */
  sup: Side;
  /** 摆动腿（= 另一条） */
  sw: Side;
  /** 前腿（几何 x，靠前的那只） */
  front: Side;
  /** 后腿 */
  rear: Side;
  /** 承接腿 = 本周期要成为承重腿的那只（DOUBLE/LOAD 期间由状态机指派） */
  recv: Side;
  /** 本拍刚触地（边沿） */
  touchdown: Record<Side, boolean>;
  /** 本拍刚离地（边沿） */
  liftoff: Record<Side, boolean>;
  /** 摆动腿膝角速度（deg/s，负 = 仍在屈曲，正 = 已在伸展） */
  swingKneeVel: number;
  /** 承重腿膝角速度（deg/s） */
  supKneeVel: number;
  /** 离地净空（m，≥0 表示已离地） */
  clearance: number;
  /** 距上次抬腿的间隔（s） */
  sinceStep: number;
  /** 帧域越界项数（由 `checkDomains` 填） */
  domainBad: number;
  /** 帧域越界总偏差（deg） */
  domainWorst: number;
}

/**

 *   新增状态或改判据**只改这一张表**，代码里不许散落字面量。

    // 两条腿都还带着重量 —— 真双支撑，而不是"一条腿已经卸掉"

      val: (c) => Math.min(c.rs.loadFrac[c.front], c.rs.loadFrac[c.rear]), tol: (c) => c.cfg.loadReleaseFrac },
    { item: '站姿在帧域内', ok: (c) => c.domainBad === 0, val: (c) => c.domainBad, tol: () => 0, hard: true },
    { item: '躯干倾角', ok: (c) => Math.abs(c.rs.tiltDeg) <= c.cfg.tiltMaxDeg,

  LOAD: Object.freeze([

    { item: '后脚仍未离地', ok: (c) => c.rs.grounded[c.rear],

/** 膝关节索引（按腿侧）。`jointIndexByName` 的轻量包装，避免每次查表 */
function J_KNEE(side: Side): number {
  const sk = KNEE_SIDE_CACHE;
  return sk ? (side === 'l' ? sk.l : sk.r) : -1;
}
let KNEE_SIDE_CACHE: { l: number; r: number } | null = null;
/** 由 `RigState` 初始化一次（`jointIndexByName` 需要骨架，rigState 里有） */
export function bindJoints(rs: RigState): void {
  const nm = (n: string): number => rs.sk.joints.findIndex((j) => j.name === n);
  KNEE_SIDE_CACHE = { l: nm('knee_l'), r: nm('knee_r') };
}
