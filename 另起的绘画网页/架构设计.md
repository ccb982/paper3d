// 检测点是否命中注释（基于几何形状）
const hitTestAnnotation = (annotation: Annotation, worldPoint: Point, threshold: number = 10): boolean => {
  const { geometry } = annotation;
  switch (geometry.type) {
    case 'point':
      return Math.hypot(worldPoint.x - geometry.points[0].x, worldPoint.y - geometry.points[0].y) < threshold;
    case 'polyline':
    case 'polygon': {
      const points = geometry.points;
      for (let i = 0; i < points.length - 1; i++) {
        const a = points[i], b = points[i+1];
        const dist = distanceToLineSegment(worldPoint.x, worldPoint.y, a.x, a.y, b.x, b.y);
        if (dist < threshold) return true;
      }
      if (geometry.type === 'polygon' && points.length >= 3) {
        // 封闭多边形内部点击也可命中（可选）
        if (isPointInPolygon(worldPoint, points)) return true;
      }
      return false;
    }
    default: return false;
  }
};

// 在 loadFromStorage 后调用
const migrateOldAnnotations = (shapes: Shape[]): Annotation[] => {
  const oldAnnotations: Annotation[] = [];
  shapes.forEach(shape => {
    if (shape.annotation) {
      // 图形中心点作为注释位置
      const center = shape.points.reduce((sum, p) => ({ x: sum.x + p.x, y: sum.y + p.y }), { x:0, y:0 });
      center.x /= shape.points.length; center.y /= shape.points.length;
      oldAnnotations.push({
        id: `migrated_${shape.id}`,
        text: shape.annotation,
        geometry: { type: 'point', points: [center] },
        layerId: shape.layerId,
        createdAt: Date.now(),
        updatedAt: Date.now(),
      });
    }
    shape.points.forEach((point, idx) => {
      if (point.annotation) {
        oldAnnotations.push({
          id: `migrated_${shape.id}_p${idx}`,
          text: point.annotation,
          geometry: { type: 'point', points: [point] },
          layerId: shape.layerId,
          createdAt: Date.now(),
          updatedAt: Date.now(),
        });
      }
    });
  });
  return oldAnnotations;
};
