import type { Point } from '../types';

// 线段求交（含端点）
export function lineIntersection(
  p1: Point, p2: Point, p3: Point, p4: Point
): Point | null {
  const d = (p2.x - p1.x) * (p4.y - p3.y) - (p2.y - p1.y) * (p4.x - p3.x);
  if (Math.abs(d) < 1e-8) return null;
  const u = ((p3.x - p1.x) * (p4.y - p3.y) - (p3.y - p1.y) * (p4.x - p3.x)) / d;
  const v = ((p3.x - p1.x) * (p2.y - p1.y) - (p3.y - p1.y) * (p2.x - p1.x)) / d;
  if (u >= 0 && u <= 1 && v >= 0 && v <= 1) {
    return { x: p1.x + u * (p2.x - p1.x), y: p1.y + u * (p2.y - p1.y) };
  }
  return null;
}

// 线段与圆求交（返回参数 t 值，线段参数化 P = P0 + t*(P1-P0)）
export function lineCircleIntersection(
  p0: Point, p1: Point, cx: number, cy: number, r: number
): number[] {
  const dx = p1.x - p0.x, dy = p1.y - p0.y;
  const fx = p0.x - cx, fy = p0.y - cy;
  const a = dx*dx + dy*dy;
  const b = 2*(fx*dx + fy*dy);
  const c = fx*fx + fy*fy - r*r;
  const delta = b*b - 4*a*c;
  if (delta < 1e-8) return [];
  const sqrtD = Math.sqrt(delta);
  const t1 = (-b - sqrtD) / (2*a);
  const t2 = (-b + sqrtD) / (2*a);
  const res: number[] = [];
  if (t1 >= 0 && t1 <= 1) res.push(t1);
  if (t2 >= 0 && t2 <= 1 && Math.abs(t2-t1) > 1e-8) res.push(t2);
  return res;
}

// 二次贝塞尔曲线与线段求交（数值迭代）
export function bezierLineIntersection(
  p0: Point, p1: Point, ctrl: Point,
  lineStart: Point, lineEnd: Point
): number[] {
  // 粗略采样后精确二分，这里简化实现，实际应使用贝塞尔-直线求交公式
  const tValues: number[] = [];
  const steps = 20;
  for (let i = 0; i < steps; i++) {
    const t = i / steps;
    const mt = 1-t;
    const x = mt*mt*p0.x + 2*mt*t*ctrl.x + t*t*p1.x;
    const y = mt*mt*p0.y + 2*mt*t*ctrl.y + t*t*p1.y;
    // 检查点是否在线段上（容差）
    const dist = distanceToLineSegment({x,y}, lineStart, lineEnd);
    if (dist < 1e-5) tValues.push(t);
  }
  // 去重
  return [...new Set(tValues.map(t => Math.round(t*1e6)/1e6))];
}

// 点到线段距离
export function distanceToLineSegment(p: Point, a: Point, b: Point): number {
  const ax = p.x - a.x, ay = p.y - a.y;
  const bx = b.x - a.x, by = b.y - a.y;
  const dot = ax*bx + ay*by;
  const len2 = bx*bx + by*by;
  if (len2 === 0) return Math.hypot(ax, ay);
  let t = dot / len2;
  t = Math.max(0, Math.min(1, t));
  const projX = a.x + t*bx;
  const projY = a.y + t*by;
  return Math.hypot(p.x - projX, p.y - projY);
}

// 从 Shape 生成原始边（未打断）
function shapeToRawEdges(shape: Shape): BoundarySegment[] {
  const edges: BoundarySegment[] = [];
  // 根据 shape.type 构造对应的 BoundarySegment
  // 这里只展示部分实现
  if (shape.type === 'line' && shape.points.length >= 2) {
    edges.push({
      id: `${shape.id}_edge`,
      curveType: 'line',
      params: { start: shape.points[0], end: shape.points[1] },
      range: { start: 0, end: 1 },
      direction: 'ccw', // 临时
      sourceShapeId: shape.id,
    });
  }
  // 矩形 -> 4条直线段
  if (shape.type === 'rectangle' && shape.points.length >= 2) {
    const p1 = shape.points[0], p2 = shape.points[1];
    const minX = Math.min(p1.x, p2.x), maxX = Math.max(p1.x, p2.x);
    const minY = Math.min(p1.y, p2.y), maxY = Math.max(p1.y, p2.y);
    const rectPoints: Point[] = [
      {x:minX,y:minY}, {x:maxX,y:minY}, {x:maxX,y:maxY}, {x:minX,y:maxY}
    ];
    for (let i = 0; i < 4; i++) {
      edges.push({
        id: `${shape.id}_edge_${i}`,
        curveType: 'line',
        params: { start: rectPoints[i], end: rectPoints[(i+1)%4] },
        range: { start: 0, end: 1 },
        direction: 'ccw',
        sourceShapeId: shape.id,
      });
    }
  }
  // 圆 -> 存储圆方程，范围 [0, 2π]
  if (shape.type === 'circle' && shape.points.length >= 2) {
    const center = shape.points[0];
    const radius = Math.hypot(shape.points[1].x - center.x, shape.points[1].y - center.y);
    edges.push({
      id: `${shape.id}_circle`,
      curveType: 'circle',
      params: { cx: center.x, cy: center.y, r: radius },
      range: { start: 0, end: 2 * Math.PI },
      direction: 'ccw',
      sourceShapeId: shape.id,
    });
  }
  // 二次贝塞尔曲线
  if (shape.type === 'quadratic' && shape.points.length >= 3) {
    edges.push({
      id: `${shape.id}_bezier`,
      curveType: 'quadratic',
      params: { p0: shape.points[0], p1: shape.points[1], ctrl: shape.points[2] },
      range: { start: 0, end: 1 },
      direction: 'ccw',
      sourceShapeId: shape.id,
    });
  }
  // 画笔 -> 多个直线段
  if (shape.type === 'brush') {
    for (let i = 0; i < shape.points.length - 1; i++) {
      edges.push({
        id: `${shape.id}_brush_${i}`,
        curveType: 'line',
        params: { start: shape.points[i], end: shape.points[i+1] },
        range: { start: 0, end: 1 },
        direction: 'ccw',
        sourceShapeId: shape.id,
      });
    }
  }
  return edges;
}

// 构建半边结构主函数
export function buildPlanarGraph(shapes: Shape[]): {
  vertices: Vertex[];
  halfEdges: HalfEdge[];
  faces: Face[];
} {
  // 1. 收集所有原始边
  let allRawEdges: BoundarySegment[] = [];
  for (const shape of shapes) {
    allRawEdges.push(...shapeToRawEdges(shape));
  }
  
  // 2. 两两求交并打断边（简化：在构建时动态打断）
  // 实际实现需要循环直到所有交点都处理完毕，这里给出逻辑框架
  let finalEdges: BoundarySegment[] = [];
  // 对每条边，与其他边求交，打断后得到新边
  for (const edge of allRawEdges) {
    const splitEdges = splitEdgeByIntersections(edge, allRawEdges);
    finalEdges.push(...splitEdges);
  }
  
  // 3. 构建顶点集合（去重，坐标容差 1e-8）
  const vertexMap = new Map<string, Vertex>();
  const getVertexKey = (p: Point) => `${Math.round(p.x*1e8)},${Math.round(p.y*1e8)}`;
  for (const edge of finalEdges) {
    // 获取起点和终点
    let startPoint: Point, endPoint: Point;
    if (edge.curveType === 'line') {
      startPoint = edge.params.start;
      endPoint = edge.params.end;
    } else if (edge.curveType === 'circle') {
      // 圆没有自然起点终点，需要根据范围采样？实际上圆是闭合的，应拆成弧段
      // 为简化，这里假设圆已经通过交点打断成多个圆弧段，每个段有明确起点和终点
      // 我们在 splitEdgeByIntersections 中对圆也做了处理
      continue;
    } // ... 其他曲线类似
    const keyStart = getVertexKey(startPoint);
    if (!vertexMap.has(keyStart)) {
      vertexMap.set(keyStart, { id: `v_${vertexMap.size}`, point: startPoint, edges: [] });
    }
    const keyEnd = getVertexKey(endPoint);
    if (!vertexMap.has(keyEnd)) {
      vertexMap.set(keyEnd, { id: `v_${vertexMap.size}`, point: endPoint, edges: [] });
    }
  }
  const vertices = Array.from(vertexMap.values());
  
  // 4. 为每条边创建两个方向的 HalfEdge
  const halfEdges: HalfEdge[] = [];
  for (const edge of finalEdges) {
    const startVert = vertexMap.get(getVertexKey(edge.params.start));
    const endVert = vertexMap.get(getVertexKey(edge.params.end));
    if (!startVert || !endVert) continue;
    const he1: HalfEdge = {
      id: `he_${halfEdges.length}`,
      start: startVert,
      end: endVert,
      curve: edge,
      twin: null as any,
      next: null as any,
      face: null,
    };
    const he2: HalfEdge = {
      id: `he_${halfEdges.length}`,
      start: endVert,
      end: startVert,
      curve: { ...edge, direction: edge.direction === 'cw' ? 'ccw' : 'cw' },
      twin: he1,
      next: null as any,
      face: null,
    };
    he1.twin = he2;
    halfEdges.push(he1, he2);
    startVert.edges.push(he1);
    endVert.edges.push(he2);
  }
  
  // 5. 计算每个顶点的出边角度排序，连接 next
  for (const vert of vertices) {
    vert.edges.sort((a,b) => {
      const angleA = Math.atan2(a.end.point.y - vert.point.y, a.end.point.x - vert.point.x);
      const angleB = Math.atan2(b.end.point.y - vert.point.y, b.end.point.x - vert.point.x);
      return angleA - angleB;
    });
    for (let i = 0; i < vert.edges.length; i++) {
      const he = vert.edges[i];
      // 从该边离开后，下一条边应该是同一顶点的前一条边的 twin？
      // 标准算法：he.next = 前一条边的 twin（角度顺序逆时针）
      const prevIndex = (i - 1 + vert.edges.length) % vert.edges.length;
      const prevHe = vert.edges[prevIndex];
      he.next = prevHe.twin;
    }
  }
  
  // 6. 遍历半边，构建面
  const faces: Face[] = [];
  const visited = new Set<string>();
  for (const he of halfEdges) {
    if (visited.has(he.id)) continue;
    // 沿着 next 走，收集组成环的半边
    const cycle: HalfEdge[] = [];
    let current = he;
    do {
      visited.add(current.id);
      cycle.push(current);
      current = current.next;
      if (!current) break;
    } while (current !== he);
    if (cycle.length === 0) continue;
    // 判断环的方向（计算面积正负）
    let area = 0;
    for (const edge of cycle) {
      area += (edge.start.point.x * edge.end.point.y - edge.end.point.x * edge.start.point.y);
    }
    const isClockwise = area < 0;
    // 创建面对象：如果是顺时针环，则为洞（内边界），否则为外边界
    // 这里简化：第一次遇到外环就创建新面，内环需要匹配到包含它的外环
    // 完整实现需要包含性测试，这里略
    const face: Face = {
      id: `face_${faces.length}`,
      outerComponent: cycle[0],
      holes: [],
      isInfinite: false, // 通过面积符号或其他方式判断外无限面
    };
    faces.push(face);
    for (const edge of cycle) edge.face = face;
  }
  
  return { vertices, halfEdges, faces };
}

import type { Point, Region, BoundarySegment } from '../types';
import type { Face, HalfEdge } from './planarGraph';

// 判断点是否在多边形内部（射线法）
function pointInPolygon(point: Point, polygon: Point[]): boolean {
  let inside = false;
  for (let i = 0, j = polygon.length-1; i < polygon.length; j = i++) {
    const xi = polygon[i].x, yi = polygon[i].y;
    const xj = polygon[j].x, yj = polygon[j].y;
    const intersect = ((yi > point.y) != (yj > point.y)) &&
      (point.x < (xj - xi) * (point.y - yi) / (yj - yi) + xi);
    if (intersect) inside = !inside;
  }
  return inside;
}

// 从半边环提取边界段列表（按顺序）
function extractBoundaryFromLoop(startHe: HalfEdge): BoundarySegment[] {
  const segments: BoundarySegment[] = [];
  let he = startHe;
  do {
    segments.push(he.curve);
    he = he.next;
  } while (he !== startHe);
  return segments;
}

// 主函数：给定点，返回包含点的区域（最内层）
export function findRegionAtPoint(point: Point, shapes: Shape[]): Region | null {
  // 构建平面图
  const { faces } = buildPlanarGraph(shapes);
  // 过滤非无限面
  const candidateFaces = faces.filter(f => !f.isInfinite);
  // 对于每个面，判断点是否在其外环内且不在任何洞内
  const containingFaces: Face[] = [];
  for (const face of candidateFaces) {
    const outerPolygon = extractBoundaryFromLoop(face.outerComponent).map(seg => {
      // 对于曲线段，需要离散化采样点做点包容测试，这里简化：取曲线上的关键点
      // 实际应转换曲线为多边形近似
      if (seg.curveType === 'line') return [seg.params.start, seg.params.end];
      // 其他曲线采样
      return [];
    }).flat();
    if (!pointInPolygon(point, outerPolygon)) continue;
    let insideHole = false;
    for (const holeHe of face.holes) {
      const holePolygon = extractBoundaryFromLoop(holeHe).map(/* 采样 */).flat();
      if (pointInPolygon(point, holePolygon)) { insideHole = true; break; }
    }
    if (!insideHole) containingFaces.push(face);
  }
  // 选择面积最小的面（最内层）
  if (containingFaces.length === 0) return null;
  containingFaces.sort((a,b) => {
    const areaA = Math.abs(computePolygonArea(extractBoundaryFromLoop(a.outerComponent)));
    const areaB = Math.abs(computePolygonArea(extractBoundaryFromLoop(b.outerComponent)));
    return areaA - areaB;
  });
  const targetFace = containingFaces[0];
  
  // 提取外边界和洞的边界段
  const outer = extractBoundaryFromLoop(targetFace.outerComponent);
  const holes = targetFace.holes.map(hole => extractBoundaryFromLoop(hole));
  // 计算包围盒
  const allPoints = outer.flatMap(seg => /* 获取端点 */);
  const bbox = {
    minX: Math.min(...allPoints.map(p=>p.x)),
    maxX: Math.max(...allPoints.map(p=>p.x)),
    minY: Math.min(...allPoints.map(p=>p.y)),
    maxY: Math.max(...allPoints.map(p=>p.y)),
  };
  return {
    id: `region_${Date.now()}`,
    outerBoundary: outer,
    innerBoundaries: holes,
    bbox,
  };
}

function computePolygonArea(points: Point[]): number {
  let area = 0;
  for (let i = 0, j = points.length-1; i < points.length; j = i++) {
    area += points[j].x * points[i].y - points[i].x * points[j].y;
  }
  return area / 2;
}

import type { Point, Shape, AxisConfig } from '../types';

// ========== 辅助几何函数 ==========
function perpendicularDistance(p: Point, a: Point, b: Point): number {
  const area = Math.abs((b.x - a.x) * (a.y - p.y) - (a.x - p.x) * (b.y - a.y));
  const len = Math.hypot(b.x - a.x, b.y - a.y);
  return len === 0 ? 0 : area / len;
}

// 简化多边形（Douglas-Peucker，保持顶点顺序）
function simplifyPolygon(points: Point[], epsilon: number): Point[] {
  if (points.length <= 3) return points;
  const closed = (points[0].x === points[points.length - 1].x && points[0].y === points[points.length - 1].y);
  let pts = closed ? points.slice(0, -1) : points;

  const indices: number[] = [0];

  function douglasPeucker(start: number, end: number) {
    let maxDist = 0;
    let index = -1;
    for (let i = start + 1; i < end; i++) {
      const dist = perpendicularDistance(pts[i], pts[start], pts[end]);
      if (dist > maxDist) {
        maxDist = dist;
        index = i;
      }
    }
    if (maxDist > epsilon) {
      douglasPeucker(start, index);
      indices.push(index);
      douglasPeucker(index, end);
    }
  }

  douglasPeucker(0, pts.length - 1);
  indices.push(pts.length - 1);
  const simplified = indices.map(i => pts[i]);
  if (closed && simplified.length > 0 && 
      (simplified[0].x !== simplified[simplified.length - 1].x || 
       simplified[0].y !== simplified[simplified.length - 1].y)) {
    simplified.push(simplified[0]);
  }
  return simplified;
}

// 绘制闭合形状到掩膜
function renderShapesToMask(
  shapes: Shape[],
  worldBounds: { xMin: number; xMax: number; yMin: number; yMax: number },
  maskWidth: number,
  maskHeight: number
): ImageData {
  const canvas = document.createElement('canvas');
  canvas.width = maskWidth;
  canvas.height = maskHeight;
  const ctx = canvas.getContext('2d')!;
  ctx.fillStyle = '#000000';
  ctx.fillRect(0, 0, maskWidth, maskHeight);
  ctx.fillStyle = '#ffffff';

  const worldToPixel = (x: number, y: number) => {
    const px = ((x - worldBounds.xMin) / (worldBounds.xMax - worldBounds.xMin)) * maskWidth;
    const py = ((worldBounds.yMax - y) / (worldBounds.yMax - worldBounds.yMin)) * maskHeight;
    return { x: px, y: py };
  };

  const drawPolygon = (points: Point[]) => {
    if (points.length < 3) return;
    ctx.beginPath();
    const first = worldToPixel(points[0].x, points[0].y);
    ctx.moveTo(first.x, first.y);
    for (let i = 1; i < points.length; i++) {
      const p = worldToPixel(points[i].x, points[i].y);
      ctx.lineTo(p.x, p.y);
    }
    ctx.closePath();
    ctx.fill();
  };

  const drawCircle = (center: Point, radius: number) => {
    const c = worldToPixel(center.x, center.y);
    const scaleX = maskWidth / (worldBounds.xMax - worldBounds.xMin);
    const r = radius * scaleX;
    ctx.beginPath();
    ctx.arc(c.x, c.y, r, 0, Math.PI * 2);
    ctx.fill();
  };

  for (const shape of shapes) {
    // 仅处理闭合形状（开放形状不支持区域检测）
    if (shape.type === 'line' || shape.type === 'quadratic' || shape.type === 'brush') continue;

    if (shape.type === 'rectangle' && shape.points.length >= 2) {
      const p1 = shape.points[0];
      const p2 = shape.points[1];
      const rectPoints: Point[] = [
        { x: p1.x, y: p1.y },
        { x: p2.x, y: p1.y },
        { x: p2.x, y: p2.y },
        { x: p1.x, y: p2.y },
      ];
      drawPolygon(rectPoints);
    } else if (shape.type === 'circle' && shape.points.length >= 2) {
      drawCircle(shape.points[0], Math.hypot(shape.points[1].x - shape.points[0].x, shape.points[1].y - shape.points[0].y));
    } else if (shape.type === 'triangle' && shape.points.length >= 3) {
      drawPolygon(shape.points.slice(0, 3));
    } else if (shape.type === 'point') {
      // 点无面积，忽略
    } else {
      // 其他类型（多边形）使用 points 作为多边形顶点
      if (shape.points.length >= 3) {
        drawPolygon(shape.points);
      }
    }
  }
  return ctx.getImageData(0, 0, maskWidth, maskHeight);
}

// 泛洪填充（4邻域），返回填充像素坐标数组
function floodFill(imageData: ImageData, startX: number, startY: number): { x: number; y: number }[] {
  const { width, height, data } = imageData;
  const visited = new Uint8Array(width * height);
  const queue: [number, number][] = [[startX, startY]];
  const targetColor = data[(startY * width + startX) * 4]; // R通道
  const region: { x: number; y: number }[] = [];

  while (queue.length) {
    const [x, y] = queue.shift()!;
    const idx = y * width + x;
    if (visited[idx]) continue;
    if (data[idx * 4] !== targetColor) continue;
    visited[idx] = 1;
    region.push({ x, y });
    if (x > 0) queue.push([x - 1, y]);
    if (x < width - 1) queue.push([x + 1, y]);
    if (y > 0) queue.push([x, y - 1]);
    if (y < height - 1) queue.push([x, y + 1]);
  }
  return region;
}

// Moore-neighbor 轮廓追踪
function traceContour(mask: Uint8Array, width: number, height: number, startX: number, startY: number): { x: number; y: number }[] {
  const dirs = [
    { dx: 0, dy: -1 }, // 上
    { dx: 1, dy: -1 }, // 右上
    { dx: 1, dy: 0 },  // 右
    { dx: 1, dy: 1 },  // 右下
    { dx: 0, dy: 1 },  // 下
    { dx: -1, dy: 1 }, // 左下
    { dx: -1, dy: 0 }, // 左
    { dx: -1, dy: -1 },// 左上
  ];
  let x = startX, y = startY;
  let prevDirIndex = 0;
  const contour: { x: number; y: number }[] = [];
  do {
    contour.push({ x, y });
    let found = false;
    for (let i = 0; i < dirs.length; i++) {
      const idx = (prevDirIndex + i) % dirs.length;
      const nx = x + dirs[idx].dx;
      const ny = y + dirs[idx].dy;
      if (nx >= 0 && nx < width && ny >= 0 && ny < height && mask[ny * width + nx] === 1) {
        x = nx;
        y = ny;
        prevDirIndex = (idx + 4) % dirs.length; // 顺时针旋转方向
        found = true;
        break;
      }
    }
    if (!found) break;
  } while (!(x === startX && y === startY));
  return contour;
}

// 像素轮廓转世界坐标多边形
function pixelsToWorld(
  pixels: { x: number; y: number }[],
  worldBounds: { xMin: number; xMax: number; yMin: number; yMax: number },
  maskSize: { width: number; height: number }
): Point[] {
  const points: Point[] = [];
  for (const p of pixels) {
    const worldX = worldBounds.xMin + (p.x / maskSize.width) * (worldBounds.xMax - worldBounds.xMin);
    const worldY = worldBounds.yMax - (p.y / maskSize.height) * (worldBounds.yMax - worldBounds.yMin);
    points.push({ x: worldX, y: worldY });
  }
  // 确保闭合
  if (points.length > 0 && (points[0].x !== points[points.length - 1].x || points[0].y !== points[points.length - 1].y)) {
    points.push(points[0]);
  }
  return points;
}

/**
 * 基于泛洪填充的区域检测
 * @param worldPoint 鼠标点击的世界坐标
 * @param shapes 当前图层的所有形状（传入前需按图层过滤）
 * @param axis 坐标轴范围
 * @param maskResolution 掩膜分辨率（建议512或1024）
 * @returns 多边形顶点和包围盒，若点击区域无效则返回null
 */
export function detectRegionByFloodFill(
  worldPoint: Point,
  shapes: Shape[],
  axis: AxisConfig,
  maskResolution: number = 512
): { polygon: Point[]; bbox: { minX: number; maxX: number; minY: number; maxY: number } } | null {
  const worldBounds = { xMin: axis.xMin, xMax: axis.xMax, yMin: axis.yMin, yMax: axis.yMax };
  const maskWidth = maskResolution;
  const maskHeight = maskResolution;

  // 1. 渲染形状到掩膜
  const imageData = renderShapesToMask(shapes, worldBounds, maskWidth, maskHeight);

  // 2. 计算点击点的像素坐标
  let pixelX = ((worldPoint.x - worldBounds.xMin) / (worldBounds.xMax - worldBounds.xMin)) * maskWidth;
  let pixelY = ((worldBounds.yMax - worldPoint.y) / (worldBounds.yMax - worldBounds.yMin)) * maskHeight;
  pixelX = Math.min(Math.max(pixelX, 0), maskWidth - 1);
  pixelY = Math.min(Math.max(pixelY, 0), maskHeight - 1);

  // 检查是否在白色区域内部
  const idx = (Math.floor(pixelY) * maskWidth + Math.floor(pixelX)) * 4;
  if (imageData.data[idx] === 0) return null;

  // 3. 泛洪填充得到区域像素
  const regionPixels = floodFill(imageData, Math.floor(pixelX), Math.floor(pixelY));
  if (regionPixels.length === 0) return null;

  // 4. 构建二值掩膜用于轮廓提取
  const mask = new Uint8Array(maskWidth * maskHeight);
  for (const px of regionPixels) {
    mask[px.y * maskWidth + px.x] = 1;
  }

  // 5. 寻找轮廓起点（区域内任意边界点）
  let startX = -1, startY = -1;
  for (let y = 0; y < maskHeight; y++) {
    for (let x = 0; x < maskWidth; x++) {
      if (mask[y * maskWidth + x] === 1) {
        // 检查是否为边界（四邻域存在0）
        let isBoundary = false;
        if (x === 0 || x === maskWidth - 1 || y === 0 || y === maskHeight - 1) {
          isBoundary = true;
        } else {
          if (mask[y * maskWidth + (x - 1)] === 0 || mask[y * maskWidth + (x + 1)] === 0 ||
              mask[(y - 1) * maskWidth + x] === 0 || mask[(y + 1) * maskWidth + x] === 0) {
            isBoundary = true;
          }
        }
        if (isBoundary) {
          startX = x;
          startY = y;
          break;
        }
      }
    }
    if (startX !== -1) break;
  }
  if (startX === -1) return null;

  // 6. 轮廓追踪
  let contourPixels = traceContour(mask, maskWidth, maskHeight, startX, startY);
  if (contourPixels.length < 3) return null;

  // 7. 转换为世界坐标并简化多边形
  let polygon = pixelsToWorld(contourPixels, worldBounds, { width: maskWidth, height: maskHeight });
  polygon = simplifyPolygon(polygon, 0.005); // 容差可调

  // 8. 计算包围盒
  const xs = polygon.map(p => p.x);
  const ys = polygon.map(p => p.y);
  const bbox = {
    minX: Math.min(...xs),
    maxX: Math.max(...xs),
    minY: Math.min(...ys),
    maxY: Math.max(...ys),
  };
  return { polygon, bbox };
}

function isPointInPolygon(point: Point, polygon: Point[]): boolean {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const xi = polygon[i].x, yi = polygon[i].y;
    const xj = polygon[j].x, yj = polygon[j].y;
    const intersect = ((yi > point.y) !== (yj > point.y)) &&
      (point.x < (xj - xi) * (point.y - yi) / (yj - yi) + xi);
    if (intersect) inside = !inside;
  }
  return inside;
}

5. 测试与集成说明
类型检查：确保 types.ts 中已经导出 Point, Shape, AxisConfig 等基础类型。

Store 导入：在 useAppStore.ts 中导入新的注释类型，并确保 addAnnotation 重载正确。

区域检测行为：

选择注释工具（📝）。

点击任意闭合形状（矩形、圆形、三角形或多边形）内部，会自动生成半透明蓝色区域，弹出文本输入框。

保存后，区域注释会固定在画布上，可再次点击内部进行编辑。

边缘情况：

点击开放形状（线、画笔、贝塞尔）内部不会生成区域注释，而是创建点注释。

若区域非常细小，可调整 maskResolution 为 1024 提高精度（注意性能）。

多边形简化容差 0.005 可根据实际轴范围调整（例如轴范围 0~100 时，容差可设为 0.01）。




方案设计：基于预计算线段集的洪水填充
你提出的思路很合理：提前将每个图形离散成线段集合，存储起来；洪水填充时直接对这些线段做碰撞检测，而不是每次都重新渲染离屏 canvas。这可以避免渲染相关的各种问题（背景色、填充与描边混淆、线宽影响等），让边界判定更加可控和精确。

下面我详细说明这个方案的设计要点。

一、核心数据结构
为每个图形存储一个 线段集（Segments），线段由两个端点 (x1, y1) → (x2, y2) 在世界坐标系中定义。

ts
interface ShapeSegments {
  shapeId: string;
  segments: Array<{ p1: Point; p2: Point }>;
  // 可选：如果是封闭图形，可以标记 isClosed: true
}
存储位置：可以放在 Shape 对象的一个缓存字段中，或者单独维护一个 Map<shapeId, segments>。

何时生成：

创建新图形时立即生成其线段集。

编辑图形（移动点、调整大小）时更新对应线段集。

删除图形时移除线段集。

这样填充时就不需要重新遍历所有图形的绘制逻辑，直接读取预计算的线段。

二、线段近似规则（按图形类型）
图形类型	离散方式
点	不产生线段（或忽略）
线段	直接取起点和终点为一条线段
矩形	4 条边（左上→右上，右上→右下，右下→左下，左下→左上）
圆	采样 N 条线段（例如 32~64 条），首尾相连形成封闭多边形
三角形	3 条边
贝塞尔曲线（二次）	采样 M 个点，连接成折线段（M 根据曲线长度自适应）
画笔（自由笔刷）	将记录的点序列连接成折线段（相邻点之间）
关键：所有线段都是世界坐标下的，保持原始精度，不需要考虑视图缩放。

三、洪水填充算法（基于线段集）
不再使用离屏 canvas 像素，而是直接在几何空间内进行填充。

1. 网格化加速（可选）
由于世界坐标是连续的，直接做几何填充比较困难。通常做法是：

将世界坐标范围（由 axis 定义）离散成一个 网格（例如 512×512 或更高）。

每个网格单元标记是否被线段穿过或阻挡。

然后在网格上进行洪水填充。

2. 碰撞检测（线段 vs 填充点）
填充时，判断一个网格单元是否可以填充：

检查该单元中心点与所有线段的最短距离。

若距离小于阈值（例如网格半宽），则该单元被阻挡，不能填充。

否则为可填充区域。

为了提高性能，可以预先构建 空间索引（如四叉树或网格哈希），快速筛选可能碰撞的线段。

3. 填充步骤
将鼠标点击的世界坐标映射到最近的网格单元作为种子。

如果种子单元被阻挡（点击在线段上），则向周围偏移寻找第一个非阻挡单元。

使用栈/队列进行四方向或八方向填充，只填充非阻挡单元。

填充完成后，获得一组连续的网格单元。

4. 提取边界多边形
对填充后的网格区域运行 Marching Squares 算法，得到平滑的轮廓多边形（网格级别）。

将轮廓顶点从网格坐标映射回世界坐标。

四、处理拐角和封闭性
封闭性保证
因为所有图形都预先生成了连续的线段（圆、矩形等封闭图形首尾相连），线段集自然定义了闭合的障碍物边界。只要填充时不跨越这些线段，区域就能正确封闭。

拐角处理
线段端点连接处可能存在微小间隙（由于浮点误差或采样不足）。需要设置一个连接容差：如果两条线段的端点距离小于 epsilon（例如 1e-6），则视为相连。

填充时的阻挡判断使用 点到线段距离，并考虑一个小的缓冲区（例如网格半宽），这样即使线段端点不完全重合，也能阻挡填充通过，从而保持封闭。

五、优点与缺点
优点
可靠：不受渲染器（抗锯齿、线宽、填充色）影响，边界判定完全由几何计算保证。

性能可控：预计算线段只需在图形变化时更新；填充时可通过降采样网格来平衡精度与速度。

易于调试：可以将所有线段可视化绘制到画布上，方便检查是否连续、封闭。

支持任意形状：只要能将形状离散为线段，就能作为边界。

缺点
实现复杂度较高：需要维护线段集、构建空间索引、实现基于网格的洪水填充。

内存占用：每个图形存储若干线段，复杂图形（画笔）可能产生大量线段。

精度权衡：网格分辨率决定了边界的平滑度，需要根据 axis 范围和最小特征尺寸动态调整。

import { Point, Shape } from '../types';

export interface Segment {
  p1: Point;
  p2: Point;
}

const segmentCache = new Map<string, Segment[]>();

function distanceToPoint(px: number, py: number, qx: number, qy: number): number {
  return Math.hypot(px - qx, py - qy);
}

function sampleQuadraticCurve(p0: Point, p1: Point, ctrl: Point, segments = 30): Point[] {
  const points: Point[] = [];
  for (let i = 0; i <= segments; i++) {
    const t = i / segments;
    const mt = 1 - t;
    const x = mt * mt * p0.x + 2 * mt * t * ctrl.x + t * t * p1.x;
    const y = mt * mt * p0.y + 2 * mt * t * ctrl.y + t * t * p1.y;
    points.push({ x, y });
  }
  return points;
}

function generateSegmentsForShape(shape: Shape): Segment[] {
  const segments: Segment[] = [];
  const points = shape.points;
  if (points.length < 2) return segments;

  switch (shape.type) {
    case 'point':
      // 点不产生线段
      break;
    case 'line':
      if (points.length >= 2) {
        segments.push({ p1: points[0], p2: points[1] });
      }
      break;
    case 'rectangle':
      if (points.length >= 2) {
        const p1 = points[0];
        const p2 = points[1];
        const minX = Math.min(p1.x, p2.x);
        const maxX = Math.max(p1.x, p2.x);
        const minY = Math.min(p1.y, p2.y);
        const maxY = Math.max(p1.y, p2.y);
        segments.push(
          { p1: { x: minX, y: minY }, p2: { x: maxX, y: minY } },
          { p1: { x: maxX, y: minY }, p2: { x: maxX, y: maxY } },
          { p1: { x: maxX, y: maxY }, p2: { x: minX, y: maxY } },
          { p1: { x: minX, y: maxY }, p2: { x: minX, y: minY } }
        );
      }
      break;
    case 'circle':
      if (points.length >= 2) {
        const center = points[0];
        const radius = Math.hypot(points[1].x - center.x, points[1].y - center.y);
        const steps = 64;
        const vertices: Point[] = [];
        for (let i = 0; i <= steps; i++) {
          const angle = (i / steps) * Math.PI * 2;
          vertices.push({
            x: center.x + radius * Math.cos(angle),
            y: center.y + radius * Math.sin(angle),
          });
        }
        for (let i = 0; i < vertices.length - 1; i++) {
          segments.push({ p1: vertices[i], p2: vertices[i + 1] });
        }
      }
      break;
    case 'triangle':
      if (points.length >= 3) {
        segments.push(
          { p1: points[0], p2: points[1] },
          { p1: points[1], p2: points[2] },
          { p1: points[2], p2: points[0] }   // 闭合
        );
      }
      break;
    case 'quadratic':
      if (points.length >= 3) {
        const sampled = sampleQuadraticCurve(points[0], points[1], points[2], 40);
        for (let i = 0; i < sampled.length - 1; i++) {
          segments.push({ p1: sampled[i], p2: sampled[i + 1] });
        }
      }
      break;
    case 'brush':
      for (let i = 0; i < points.length - 1; i++) {
        segments.push({ p1: points[i], p2: points[i + 1] });
      }
      break;
    default:
      // 通用多边形（如多边形工具或自定义闭合形状）
      for (let i = 0; i < points.length - 1; i++) {
        segments.push({ p1: points[i], p2: points[i + 1] });
      }
      // 如果形状标记为闭合或为多边形，则闭合首尾
      if (shape.closed || shape.type === 'polygon') {
        segments.push({ p1: points[points.length - 1], p2: points[0] });
      }
      break;
  }
  return segments;
}

// ========== 对外 API ==========
export function updateShapeSegments(shape: Shape): void {
  const segments = generateSegmentsForShape(shape);
  segmentCache.set(shape.id, segments);
}

export function removeShapeSegments(shapeId: string): void {
  segmentCache.delete(shapeId);
}

export function clearAllSegments(): void {
  segmentCache.clear();
}

export function getAllSegmentsForLayer(layerId: string, shapes: Shape[]): Segment[] {
  const allSegments: Segment[] = [];
  for (const shape of shapes) {
    if (shape.layerId !== layerId) continue;
    let segments = segmentCache.get(shape.id);
    if (!segments) {
      segments = generateSegmentsForShape(shape);
      segmentCache.set(shape.id, segments);
    }
    allSegments.push(...segments);
  }
  return allSegments;
}

import { Point, AxisConfig } from '../types';
import type { Segment } from './segmentCache';

// ========== 几何辅助 ==========
function distanceToLineSegment(px: number, py: number, x1: number, y1: number, x2: number, y2: number): number {
  const ax = px - x1, ay = py - y1;
  const bx = x2 - x1, by = y2 - y1;
  const dot = ax * bx + ay * by;
  const len2 = bx * bx + by * by;
  if (len2 === 0) return Math.hypot(ax, ay);
  let t = dot / len2;
  t = Math.max(0, Math.min(1, t));
  const projX = x1 + t * bx;
  const projY = y1 + t * by;
  return Math.hypot(px - projX, py - projY);
}

// ========== 标准 Marching Squares 轮廓提取 ==========
// 根据 2x2 网格状态返回边界段 (采用经典查表法)
const MARCHING_SQUARES_EDGES: Record<number, [number, number][]> = {
  0: [],
  1: [[0, 1]],
  2: [[1, 2]],
  3: [[0, 2]],
  4: [[2, 3]],
  5: [[0, 1], [2, 3]],
  6: [[1, 3]],
  7: [[0, 3]],
  8: [[0, 3]],
  9: [[1, 3]],
  10: [[0, 1], [2, 3]],
  11: [[2, 3]],
  12: [[0, 2]],
  13: [[1, 2]],
  14: [[0, 1]],
  15: [],
};

// 将像素网格轮廓转换为连续多边形（简单连接）
function polygonizeContour(segments: [number, number][], gridW: number, gridH: number): [number, number][] {
  if (segments.length === 0) return [];
  // 将线段转换为邻接表
  const adj = new Map<string, [number, number][]>();
  const addEdge = (a: [number, number], b: [number, number]) => {
    const key = `${a[0]},${a[1]}`;
    if (!adj.has(key)) adj.set(key, []);
    adj.get(key)!.push(b);
  };
  for (const [start, end] of segments) {
    const p1 = [start & 1, start >> 1] as [number, number];
    const p2 = [end & 1, end >> 1] as [number, number];
    addEdge(p1, p2);
    addEdge(p2, p1);
  }
  if (adj.size === 0) return [];
  // 贪心连接成环
  const startKey = Array.from(adj.keys())[0];
  const start = startKey.split(',').map(Number) as [number, number];
  const polygon: [number, number][] = [start];
  let current = start;
  let prev: [number, number] | null = null;
  const visited = new Set<string>();
  visited.add(`${current[0]},${current[1]}`);
  while (true) {
    const neighbors = adj.get(`${current[0]},${current[1]}`) || [];
    let next: [number, number] | null = null;
    for (const n of neighbors) {
      const key = `${n[0]},${n[1]}`;
      if (!prev || (n[0] !== prev[0] || n[1] !== prev[1])) {
        next = n;
        break;
      }
    }
    if (!next) break;
    const nextKey = `${next[0]},${next[1]}`;
    if (visited.has(nextKey)) {
      // 闭环检测
      if (next[0] === start[0] && next[1] === start[1]) break;
      else break;
    }
    visited.add(nextKey);
    polygon.push(next);
    prev = current;
    current = next;
  }
  // 转换坐标：将 (gx, gy) 映射到 (x, y) 将在外部处理
  return polygon;
}

function marchingSquares(filled: boolean[][], width: number, height: number): [number, number][] {
  if (width < 2 || height < 2) return [];
  const edgeSegments: [number, number][] = [];
  for (let y = 0; y < height - 1; y++) {
    for (let x = 0; x < width - 1; x++) {
      const idx = (filled[y][x] ? 1 : 0)
                | (filled[y][x+1] ? 2 : 0)
                | (filled[y+1][x+1] ? 4 : 0)
                | (filled[y+1][x] ? 8 : 0);
      const edges = MARCHING_SQUARES_EDGES[idx];
      if (!edges) continue;
      for (const [e1, e2] of edges) {
        // 将边缘编码映射到顶点索引 (0=左下,1=右下,2=右上,3=左上)
        // 为了简化，直接使用标准的 2x2 网格边，不需要插值，因为我们要的是离散轮廓
        const v1 = (() => {
          if (e1 === 0) return [x, y];
          if (e1 === 1) return [x+1, y];
          if (e1 === 2) return [x+1, y+1];
          return [x, y+1];
        })();
        const v2 = (() => {
          if (e2 === 0) return [x, y];
          if (e2 === 1) return [x+1, y];
          if (e2 === 2) return [x+1, y+1];
          return [x, y+1];
        })();
        // 合并到线段列表
        edgeSegments.push([v1[0] + v1[1] * width, v2[0] + v2[1] * width]);
      }
    }
  }
  // 去重并连接成多边形
  const polygonPoints = polygonizeContour(edgeSegments, width, height);
  return polygonPoints;
}

/**
 * 基于预计算线段的洪水填充区域检测（改进版）
 * @param point 世界坐标点
 * @param segments 当前图层的所有线段
 * @param axis 坐标轴范围
 * @returns 多边形和包围盒
 */
export function detectRegionByFloodFill(
  point: Point,
  segments: Segment[],
  axis: AxisConfig
): { polygon: Point[]; bbox: { minX: number; maxX: number; minY: number; maxY: number } } | null {
  if (segments.length === 0) return null;

  const { xMin, xMax, yMin, yMax } = axis;
  const worldWidth = xMax - xMin;
  const worldHeight = yMax - yMin;
  if (worldWidth <= 0 || worldHeight <= 0) return null;

  // 动态网格分辨率：保证最小单元尺寸不超过世界范围的 1/200，且范围 64~512
  const targetCellSize = Math.min(worldWidth, worldHeight) / 200;
  let gridRes = Math.floor(Math.min(worldWidth, worldHeight) / targetCellSize);
  gridRes = Math.min(512, Math.max(64, gridRes));
  const cellW = worldWidth / gridRes;
  const cellH = worldHeight / gridRes;
  const threshold = Math.min(cellW, cellH) * 0.4; // 碰撞阈值

  const worldToGrid = (wx: number, wy: number): [number, number] => {
    let gx = (wx - xMin) / worldWidth * gridRes;
    let gy = (yMax - wy) / worldHeight * gridRes;
    gx = Math.min(Math.max(0, gx), gridRes - 1);
    gy = Math.min(Math.max(0, gy), gridRes - 1);
    return [Math.floor(gx), Math.floor(gy)];
  };

  const gridCenterWorld = (gx: number, gy: number): Point => {
    const cx = xMin + (gx + 0.5) * cellW;
    const cy = yMax - (gy + 0.5) * cellH;
    return { x: cx, y: cy };
  };

  // 构建阻挡矩阵
  const blocked: boolean[][] = Array(gridRes).fill(0).map(() => Array(gridRes).fill(false));
  for (let gy = 0; gy < gridRes; gy++) {
    for (let gx = 0; gx < gridRes; gx++) {
      const center = gridCenterWorld(gx, gy);
      let isBlocked = false;
      for (const seg of segments) {
        const dist = distanceToLineSegment(center.x, center.y, seg.p1.x, seg.p1.y, seg.p2.x, seg.p2.y);
        if (dist < threshold) {
          isBlocked = true;
          break;
        }
      }
      blocked[gy][gx] = isBlocked;
    }
  }

  // 种子点搜索
  let [seedX, seedY] = worldToGrid(point.x, point.y);
  if (blocked[seedY][seedX]) {
    let found = false;
    for (let radius = 1; radius <= 5; radius++) {
      for (let dy = -radius; dy <= radius; dy++) {
        for (let dx = -radius; dx <= radius; dx++) {
          const nx = seedX + dx;
          const ny = seedY + dy;
          if (nx >= 0 && nx < gridRes && ny >= 0 && ny < gridRes && !blocked[ny][nx]) {
            seedX = nx;
            seedY = ny;
            found = true;
            break;
          }
        }
        if (found) break;
      }
      if (found) break;
    }
    if (!found) return null;
  }

  // BFS 填充
  const filled: boolean[][] = Array(gridRes).fill(0).map(() => Array(gridRes).fill(false));
  const queue: [number, number][] = [[seedX, seedY]];
  filled[seedY][seedX] = true;
  while (queue.length) {
    const [x, y] = queue.shift()!;
    const neighbors: [number, number][] = [[x+1,y], [x-1,y], [x,y+1], [x,y-1]];
    for (const [nx, ny] of neighbors) {
      if (nx >= 0 && nx < gridRes && ny >= 0 && ny < gridRes && !filled[ny][nx] && !blocked[ny][nx]) {
        filled[ny][nx] = true;
        queue.push([nx, ny]);
      }
    }
  }

  let fillCount = 0;
  for (let i = 0; i < gridRes; i++) {
    for (let j = 0; j < gridRes; j++) {
      if (filled[i][j]) fillCount++;
    }
  }
  if (fillCount < 4) return null;

  // 轮廓提取（改进的 Marching Squares）
  const contourGrid = marchingSquares(filled, gridRes, gridRes);
  if (contourGrid.length < 3) return null;

  const polygon: Point[] = contourGrid.map(([gx, gy]) => {
    const wx = xMin + (gx / gridRes) * worldWidth;
    const wy = yMax - (gy / gridRes) * worldHeight;
    return { x: wx, y: wy };
  });

  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const p of polygon) {
    minX = Math.min(minX, p.x);
    maxX = Math.max(maxX, p.x);
    minY = Math.min(minY, p.y);
    maxY = Math.max(maxY, p.y);
  }

  return { polygon, bbox: { minX, maxX, minY, maxY } };
}

🔧 修复方案
1. 修改 regionFloodFill.ts 中的碰撞判定
原来的 threshold 是基于单元格尺寸的 0.12 倍（即 0.0006），而线段密度很高（圆有 64 条边，每条边很短），导致几乎所有的网格中心点距离线段的距离都大于该阈值，所以没有被阻挡。

解决方案：增大阈值，同时改用 四连通单元的四角采样 来判定是否被线段穿过，而不是仅检测中心点。

import { Point, AxisConfig } from '../types';
import type { Segment } from './segmentCache';

function distanceToLineSegment(px: number, py: number, x1: number, y1: number, x2: number, y2: number): number {
  const ax = px - x1, ay = py - y1;
  const bx = x2 - x1, by = y2 - y1;
  const len2 = bx * bx + by * by;
  if (len2 === 0) return Math.hypot(ax, ay);
  let t = (ax * bx + ay * by) / len2;
  t = Math.max(0, Math.min(1, t));
  const projX = x1 + t * bx;
  const projY = y1 + t * by;
  return Math.hypot(px - projX, py - projY);
}

function segmentIntersectsCell(seg: Segment, cellMinX: number, cellMinY: number, cellMaxX: number, cellMaxY: number, threshold: number): boolean {
  // 快速 AABB 剔除
  const segMinX = Math.min(seg.p1.x, seg.p2.x);
  const segMaxX = Math.max(seg.p1.x, seg.p2.x);
  const segMinY = Math.min(seg.p1.y, seg.p2.y);
  const segMaxY = Math.max(seg.p1.y, seg.p2.y);
  if (cellMaxX < segMinX || cellMinX > segMaxX || cellMaxY < segMinY || cellMinY > segMaxY) return false;

  // 检查四个角点到线段的距离
  const corners = [
    [cellMinX, cellMinY], [cellMaxX, cellMinY],
    [cellMaxX, cellMaxY], [cellMinX, cellMaxY]
  ];
  for (const [cx, cy] of corners) {
    if (distanceToLineSegment(cx, cy, seg.p1.x, seg.p1.y, seg.p2.x, seg.p2.y) < threshold) {
      return true;
    }
  }
  // 可选：检查线段的中点是否在单元格内（避免细长线段穿过单元格但角点距离远的情况）
  const midX = (seg.p1.x + seg.p2.x) / 2;
  const midY = (seg.p1.y + seg.p2.y) / 2;
  if (midX >= cellMinX && midX <= cellMaxX && midY >= cellMinY && midY <= cellMaxY) {
    return true;
  }
  return false;
}

export function detectRegionByFloodFill(
  point: Point,
  segments: Segment[],
  axis: AxisConfig
): { polygon: Point[]; bbox: { minX: number; maxX: number; minY: number; maxY: number } } | null {
  if (segments.length === 0) return null;

  const { xMin, xMax, yMin, yMax } = axis;
  const worldWidth = xMax - xMin;
  const worldHeight = yMax - yMin;
  if (worldWidth <= 1e-6 || worldHeight <= 1e-6) return null;

  // 动态分辨率：保证至少 100x100，最多 400x400（性能与精度平衡）
  let gridRes = Math.floor(Math.min(worldWidth, worldHeight) / (Math.min(worldWidth, worldHeight) / 200));
  gridRes = Math.min(400, Math.max(100, gridRes));
  const cellW = worldWidth / gridRes;
  const cellH = worldHeight / gridRes;
  const threshold = Math.min(cellW, cellH) * 0.25; // 提高阈值，更容易阻塞

  // 构建阻挡矩阵
  const blocked: boolean[][] = Array(gridRes).fill(0).map(() => Array(gridRes).fill(false));
  for (let gy = 0; gy < gridRes; gy++) {
    for (let gx = 0; gx < gridRes; gx++) {
      const cellMinX = xMin + gx * cellW;
      const cellMaxX = cellMinX + cellW;
      const cellMinY = yMax - (gy + 1) * cellH;
      const cellMaxY = cellMinY + cellH;
      for (const seg of segments) {
        if (segmentIntersectsCell(seg, cellMinX, cellMinY, cellMaxX, cellMaxY, threshold)) {
          blocked[gy][gx] = true;
          break;
        }
      }
    }
  }

  // 种子点映射（世界坐标到网格）
  const worldToGrid = (wx: number, wy: number): [number, number] => {
    let gx = (wx - xMin) / worldWidth * gridRes;
    let gy = (yMax - wy) / worldHeight * gridRes;
    gx = Math.min(Math.max(0, gx), gridRes - 1);
    gy = Math.min(Math.max(0, gy), gridRes - 1);
    return [Math.floor(gx), Math.floor(gy)];
  };

  let [seedX, seedY] = worldToGrid(point.x, point.y);
  // 如果种子点被阻塞，向四周搜索最近的非阻塞单元格
  if (blocked[seedY][seedX]) {
    let found = false;
    for (let r = 1; r <= 8; r++) {
      for (let dy = -r; dy <= r; dy++) {
        for (let dx = -r; dx <= r; dx++) {
          const nx = seedX + dx, ny = seedY + dy;
          if (nx >= 0 && nx < gridRes && ny >= 0 && ny < gridRes && !blocked[ny][nx]) {
            seedX = nx; seedY = ny;
            found = true;
            break;
          }
        }
        if (found) break;
      }
      if (found) break;
    }
    if (!found) return null; // 全被阻塞
  }

  // BFS 填充
  const filled: boolean[][] = Array(gridRes).fill(0).map(() => Array(gridRes).fill(false));
  const queue: [number, number][] = [[seedX, seedY]];
  filled[seedY][seedX] = true;
  while (queue.length) {
    const [x, y] = queue.shift()!;
    const neighbors: [number, number][] = [[x+1,y], [x-1,y], [x,y+1], [x,y-1]];
    for (const [nx, ny] of neighbors) {
      if (nx >= 0 && nx < gridRes && ny >= 0 && ny < gridRes && !filled[ny][nx] && !blocked[ny][nx]) {
        filled[ny][nx] = true;
        queue.push([nx, ny]);
      }
    }
  }

  // 计算填充数量，如果太小则放弃
  let fillCount = 0;
  for (let i = 0; i < gridRes; i++) {
    for (let j = 0; j < gridRes; j++) {
      if (filled[i][j]) fillCount++;
    }
  }
  if (fillCount < 6) return null;

  // 轮廓提取 (Marching Squares)
  const contour = extractContour(filled, gridRes, gridRes);
  if (contour.length < 3) return null;

  // 将轮廓网格坐标转换回世界坐标
  const polygon: Point[] = contour.map(([gx, gy]) => {
    const wx = xMin + (gx / gridRes) * worldWidth;
    const wy = yMax - (gy / gridRes) * worldHeight;
    return { x: wx, y: wy };
  });

  // 计算包围盒
  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const p of polygon) {
    minX = Math.min(minX, p.x);
    maxX = Math.max(maxX, p.x);
    minY = Math.min(minY, p.y);
    maxY = Math.max(maxY, p.y);
  }

  return { polygon, bbox: { minX, maxX, minY, maxY } };
}

// Marching Squares 标准实现（增强版，能处理边界开放情况）
function extractContour(filled: boolean[][], width: number, height: number): [number, number][] {
  if (width < 2 || height < 2) return [];

  const edgeTable: Record<number, [number, number][]> = {
    0: [], 1: [[0,1]], 2: [[1,2]], 3: [[0,2]], 4: [[2,3]],
    5: [[0,1],[2,3]], 6: [[1,3]], 7: [[0,3]], 8: [[0,3]],
    9: [[1,3]], 10: [[0,1],[2,3]], 11: [[2,3]], 12: [[0,2]],
    13: [[1,2]], 14: [[0,1]], 15: [],
  };

  const segments: [number, number][] = [];
  for (let y = 0; y < height - 1; y++) {
    for (let x = 0; x < width - 1; x++) {
      const idx = (filled[y][x] ? 1 : 0)
                | (filled[y][x+1] ? 2 : 0)
                | (filled[y+1][x+1] ? 4 : 0)
                | (filled[y+1][x] ? 8 : 0);
      const edges = edgeTable[idx];
      for (const [e1, e2] of edges) {
        const v1 = (() => {
          if (e1 === 0) return [x, y];
          if (e1 === 1) return [x+1, y];
          if (e1 === 2) return [x+1, y+1];
          return [x, y+1];
        })();
        const v2 = (() => {
          if (e2 === 0) return [x, y];
          if (e2 === 1) return [x+1, y];
          if (e2 === 2) return [x+1, y+1];
          return [x, y+1];
        })();
        segments.push([v1[0] + v1[1] * width, v2[0] + v2[1] * width]);
      }
    }
  }

  // 连接线段为多边形（处理开放轮廓时尝试闭合）
  return connectSegments(segments, width);
}

function connectSegments(segments: [number, number][], width: number): [number, number][] {
  if (segments.length === 0) return [];

  const adj = new Map<number, number[]>();
  for (const [a, b] of segments) {
    if (!adj.has(a)) adj.set(a, []);
    if (!adj.has(b)) adj.set(b, []);
    adj.get(a)!.push(b);
    adj.get(b)!.push(a);
  }

  // 找起点（度数1的顶点，若无则取任意）
  let start = -1;
  for (const [v, neigh] of adj.entries()) {
    if (neigh.length === 1) {
      start = v;
      break;
    }
  }
  if (start === -1) start = Array.from(adj.keys())[0];

  const polygon: [number, number][] = [];
  let current = start;
  let prev = -1;
  const visited = new Set<number>();
  while (true) {
    const neighbors = adj.get(current) || [];
    let next = -1;
    for (const n of neighbors) {
      if (n !== prev) {
        next = n;
        break;
      }
    }
    if (next === -1) break;
    const key = `${current},${next}`;
    if (visited.has(key)) break;
    visited.add(key);

    const px = current % width;
    const py = Math.floor(current / width);
    polygon.push([px, py]);

    prev = current;
    current = next;
  }
  // 如果不闭合，尝试用起点闭合
  if (polygon.length > 0 && (polygon[0][0] !== current % width || polygon[0][1] !== Math.floor(current / width))) {
    polygon.push([polygon[0][0], polygon[0][1]]);
  }
  return polygon;
}

// 判断线段 AB 是否与矩形 [minX, maxX] × [minY, maxY] 相交
function segmentIntersectsRect(
  ax: number, ay: number,
  bx: number, by: number,
  minX: number, maxX: number,
  minY: number, maxY: number
): boolean {
  // 先检查端点是否在矩形内
  if ((ax >= minX && ax <= maxX && ay >= minY && ay <= maxY) ||
      (bx >= minX && bx <= maxX && by >= minY && by <= maxY)) return true;
  // 检查四条边与线段相交
  const intersectLine = (x1, y1, x2, y2) => {
    const denom = (x2 - x1) * (by - ay) - (y2 - y1) * (bx - ax);
    if (denom === 0) return false;
    const t = ((bx - ax) * (y1 - ay) - (by - ay) * (x1 - ax)) / denom;
    const u = -((x2 - x1) * (y1 - ay) - (y2 - y1) * (x1 - ax)) / denom;
    return t >= 0 && t <= 1 && u >= 0 && u <= 1;
  };
  // 矩形的四条边
  if (intersectLine(minX, minY, maxX, minY, ax, ay, bx, by)) return true;
  if (intersectLine(maxX, minY, maxX, maxY, ax, ay, bx, by)) return true;
  if (intersectLine(maxX, maxY, minX, maxY, ax, ay, bx, by)) return true;
  if (intersectLine(minX, maxY, minX, minY, ax, ay, bx, by)) return true;
  return false;
}

import { Point, AxisConfig } from '../types';
import type { Segment } from './segmentCache';

// 线段与轴对齐矩形相交检测（精确，无阈值）
function segmentIntersectsRect(
  ax: number, ay: number,
  bx: number, by: number,
  minX: number, maxX: number,
  minY: number, maxY: number
): boolean {
  // 端点检查
  if ((ax >= minX && ax <= maxX && ay >= minY && ay <= maxY) ||
      (bx >= minX && bx <= maxX && by >= minY && by <= maxY)) return true;

  const intersectLine = (x1: number, y1: number, x2: number, y2: number) => {
    const denom = (x2 - x1) * (by - ay) - (y2 - y1) * (bx - ax);
    if (denom === 0) return false;
    const t = ((bx - ax) * (y1 - ay) - (by - ay) * (x1 - ax)) / denom;
    const u = -((x2 - x1) * (y1 - ay) - (y2 - y1) * (x1 - ax)) / denom;
    return t >= 0 && t <= 1 && u >= 0 && u <= 1;
  };

  // 矩形的四条边
  if (intersectLine(minX, minY, maxX, minY)) return true;
  if (intersectLine(maxX, minY, maxX, maxY)) return true;
  if (intersectLine(maxX, maxY, minX, maxY)) return true;
  if (intersectLine(minX, maxY, minX, minY)) return true;
  return false;
}

  const worldWidth = xMax - xMin;

  const targetCellSize = Math.min(worldWidth, worldHeight) / 150;

      const cellMinX = xMin + gx * cellW;
      const cellMaxX = cellMinX + cellW;

  // 世界坐标 → 网格坐标
  const worldToGrid = (wx: number, wy: number): [number, number] => {
    let gx = (wx - xMin) / worldWidth * gridRes;
    let gy = (yMax - wy) / worldHeight * gridRes;
    gx = Math.min(Math.max(0, gx), gridRes - 1);
    gy = Math.min(Math.max(0, gy), gridRes - 1);
    return [Math.floor(gx), Math.floor(gy)];
  };

  let [seedX, seedY] = worldToGrid(point.x, point.y);

  // 如果种子点被阻塞，向四周搜索最近的非阻塞单元格
  if (blocked[seedY][seedX]) {
    let found = false;
    for (let r = 1; r <= 8; r++) {
      for (let dy = -r; dy <= r; dy++) {
        for (let dx = -r; dx <= r; dx++) {
          const nx = seedX + dx, ny = seedY + dy;
          if (nx >= 0 && nx < gridRes && ny >= 0 && ny < gridRes && !blocked[ny][nx]) {
            seedX = nx; seedY = ny;
            found = true;
            break;
          }
        }
        if (found) break;
      }
      if (found) break;
    }
    if (!found) return null; // 全被阻塞
  }

  // BFS 填充
  const filled: boolean[][] = Array(gridRes).fill(0).map(() => Array(gridRes).fill(false));
  const queue: [number, number][] = [[seedX, seedY]];
  filled[seedY][seedX] = true;
  while (queue.length) {
    const [x, y] = queue.shift()!;
    const neighbors: [number, number][] = [[x+1,y], [x-1,y], [x,y+1], [x,y-1]];
    for (const [nx, ny] of neighbors) {
      if (nx >= 0 && nx < gridRes && ny >= 0 && ny < gridRes && !filled[ny][nx] && !blocked[ny][nx]) {
        filled[ny][nx] = true;
        queue.push([nx, ny]);
      }
    }
  }

  // 统计填充数量
  let fillCount = 0;
  for (let i = 0; i < gridRes; i++) {
    for (let j = 0; j < gridRes; j++) {
      if (filled[i][j]) fillCount++;
    }
  }
  if (fillCount < 6) return null;

  // 轮廓提取（Marching Squares）
  const contour = extractContour(filled, gridRes, gridRes);
  if (contour.length < 3) return null;

  // 网格 → 世界坐标
  const polygon: Point[] = contour.map(([gx, gy]) => {
    const wx = xMin + (gx / gridRes) * worldWidth;
    const wy = yMax - (gy / gridRes) * worldHeight;
    return { x: wx, y: wy };
  });

  // 计算包围盒
  let minX = Infinity, maxX = -Infinity, minY = Infinity, maxY = -Infinity;
  for (const p of polygon) {
    minX = Math.min(minX, p.x);
    maxX = Math.max(maxX, p.x);
    minY = Math.min(minY, p.y);
    maxY = Math.max(maxY, p.y);
  }

  return { polygon, bbox: { minX, maxX, minY, maxY } };
}

// ========== Marching Squares 相关函数（与原实现相同，保持稳定） ==========
const MARCHING_SQUARES_EDGES: Record<number, [number, number][]> = {
  0: [], 1: [[0,1]], 2: [[1,2]], 3: [[0,2]], 4: [[2,3]],
  5: [[0,1],[2,3]], 6: [[1,3]], 7: [[0,3]], 8: [[0,3]],
  9: [[1,3]], 10: [[0,1],[2,3]], 11: [[2,3]], 12: [[0,2]],
  13: [[1,2]], 14: [[0,1]], 15: [],
};

function polygonizeContour(segments: [number, number][], width: number): [number, number][] {
  if (segments.length === 0) return [];
  const adj = new Map<string, [number, number][]>();
  const addEdge = (a: [number, number], b: [number, number]) => {
    const key = `${a[0]},${a[1]}`;
    if (!adj.has(key)) adj.set(key, []);
    adj.get(key)!.push(b);
  };
  for (const [s, e] of segments) {
    const p1 = [s & 1, s >> 1] as [number, number];
    const p2 = [e & 1, e >> 1] as [number, number];
    addEdge(p1, p2);
    addEdge(p2, p1);
  }
  if (adj.size === 0) return [];
  const startKey = Array.from(adj.keys())[0];
  const start = startKey.split(',').map(Number) as [number, number];
  const polygon: [number, number][] = [start];
  let current = start;
  let prev: [number, number] | null = null;
  const visited = new Set<string>(startKey);
  while (true) {
    const neighbors = adj.get(`${current[0]},${current[1]}`) || [];
    let next: [number, number] | null = null;
    for (const n of neighbors) {
      const key = `${n[0]},${n[1]}`;
      if (!prev || (n[0] !== prev[0] || n[1] !== prev[1])) {
        next = n;
        break;
      }
    }
    if (!next) break;
    const nextKey = `${next[0]},${next[1]}`;
    if (visited.has(nextKey)) {
      if (next[0] === start[0] && next[1] === start[1]) break;
      else break;
    }
    visited.add(nextKey);
    polygon.push(next);
    prev = current;
    current = next;
  }
  return polygon;
}

function extractContour(filled: boolean[][], width: number, height: number): [number, number][] {
  if (width < 2 || height < 2) return [];
  const edgeSegments: [number, number][] = [];
  for (let y = 0; y < height - 1; y++) {
    for (let x = 0; x < width - 1; x++) {
      const idx = (filled[y][x] ? 1 : 0)
                | (filled[y][x+1] ? 2 : 0)
                | (filled[y+1][x+1] ? 4 : 0)
                | (filled[y+1][x] ? 8 : 0);
      const edges = MARCHING_SQUARES_EDGES[idx];
      if (!edges) continue;
      for (const [e1, e2] of edges) {
        const v1 = (() => {
          if (e1 === 0) return [x, y];
          if (e1 === 1) return [x+1, y];
          if (e1 === 2) return [x+1, y+1];
          return [x, y+1];
        })();
        const v2 = (() => {
          if (e2 === 0) return [x, y];
          if (e2 === 1) return [x+1, y];
          if (e2 === 2) return [x+1, y+1];
          return [x, y+1];
        })();
        edgeSegments.push([v1[0] + v1[1] * width, v2[0] + v2[1] * width]);
      }
    }
  }
  return polygonizeContour(edgeSegments, width);
}
现在能处理「两个图形边的交界处」吗？
能。因为我们将所有图形的线段合并后统一构建阻塞矩阵，任何由这些线段围成的封闭区域（无论形状简单还是复杂）都会被 BFS 精确填充，轮廓提取得到的就是该区域的多边形边界。

例如：

两个矩形相交，中间形成一个“十字”空洞，点击空洞内部 → 得到空洞的轮廓多边形。

一个圆形被一条直线穿过，点击左侧半圆区域 → 得到左侧半圆的轮廓。

需要确保你的 segmentCache.ts 已经为所有曲线图形（圆、贝塞尔）生成了足够密集的线段采样（你已有 64 条边，已经足够）。

注释功能彻底重构方案（点注释 + 区域注释，基于 polygon-clipping）
目标
点注释：点击任意位置直接添加文本注释，与图形无关。

区域注释：基于图形闭合区域（由矩形、圆形、三角形、闭合贝塞尔曲线、闭合画笔等组合而成）检测点击所在的区域，在该区域上添加注释。区域可以带孔（如环形区域）。

区域检测核心：每个图形在创建/更新时生成近似多边形（世界坐标闭合点集），利用 polygon-clipping 库对整个图层的所有图形多边形进行布尔运算，计算出所有独立区域（带孔多边形）。点击时根据鼠标位置查找所属区域，创建注释。

三、近似多边形生成工具（utils/approximatePolygon.ts）
为每种图形类型生成用于区域检测的闭合点集（世界坐标）。只有闭合图形才生成有效多边形，开放曲线（线段、开放贝塞尔、开放画笔）返回 []。

// utils/approximatePolygon.ts
import { Point, Shape } from '../types';

/**
 * 判断图形是否闭合（首尾点距离小于阈值）
 */
function isClosedShape(shape: Shape): boolean {
  if (shape.type === 'rectangle' || shape.type === 'circle' || shape.type === 'triangle') {
    return true;
  }
  if (shape.type === 'quadratic' || shape.type === 'brush') {
    if (shape.points.length < 3) return false;
    const first = shape.points[0];
    const last = shape.points[shape.points.length - 1];
    return Math.hypot(first.x - last.x, first.y - last.y) < 1e-6;
  }
  return false;
}

export function computeApproximatePolygon(shape: Shape): Point[] {
  if (!isClosedShape(shape)) return [];

  switch (shape.type) {
    case 'rectangle': {
      const [p1, p2] = shape.points;
      return [
        { x: p1.x, y: p1.y },
        { x: p2.x, y: p1.y },
        { x: p2.x, y: p2.y },
        { x: p1.x, y: p2.y },
        { x: p1.x, y: p1.y }, // 闭合
      ];
    }
    case 'circle': {
      const center = shape.points[0];
      const radius = Math.hypot(shape.points[1].x - center.x, shape.points[1].y - center.y);
      const segments = 48; // 足够平滑
      const points: Point[] = [];
      for (let i = 0; i <= segments; i++) {
        const angle = (i / segments) * Math.PI * 2;
        points.push({
          x: center.x + radius * Math.cos(angle),
          y: center.y + radius * Math.sin(angle),
        });
      }
      return points;
    }
    case 'triangle': {
      const [p1, p2, p3] = shape.points;
      return [p1, p2, p3, p1];
    }
    case 'quadratic': {
      // 二次贝塞尔曲线，采样足够多点构成闭合多边形
      const [p0, p1, ctrl] = shape.points; // p0起点, p1终点, ctrl控制点
      const segments = 40;
      const points: Point[] = [];
      for (let i = 0; i <= segments; i++) {
        const t = i / segments;
        const mt = 1 - t;
        const x = mt * mt * p0.x + 2 * mt * t * ctrl.x + t * t * p1.x;
        const y = mt * mt * p0.y + 2 * mt * t * ctrl.y + t * t * p1.y;
        points.push({ x, y });
      }
      return points; // 首尾已闭合（首尾点相同或接近）
    }
    case 'brush': {
      // 画笔闭合路径：直接使用所有点，并闭合
      const points = [...shape.points];
      if (points.length < 3) return [];
      // 确保首尾连接
      const first = points[0];
      const last = points[points.length - 1];
      if (Math.hypot(first.x - last.x, first.y - last.y) > 1e-6) {
        points.push(first);
      }
      return points;
    }
    default:
      return [];
  }
}

// utils/regionDetection.ts
import polygonClipping from 'polygon-clipping';
import { Point } from '../types';

/**
 * 将 Point[] 转换为 polygon-clipping 接受的格式: [[[x,y], [x,y], ...]]
 */
function toClippingFormat(polygon: Point[]): number[][][] {
  return [polygon.map(p => [p.x, p.y])];
}

/**
 * 计算图层所有近似多边形划分出的独立区域（背景区域）。
 * 算法：求出所有多边形的并集，然后用世界边界矩形减去该并集，得到剩余背景区域（每个区域可能带孔）。
 * @param polygons 每个多边形是闭合点集（Point[]），要求简单多边形（不自交）。
 * @param worldBounds 世界坐标系边界 { xMin, xMax, yMin, yMax }
 * @returns 区域列表，每个区域是 Point[][]（多环：第一个环外环，后续内洞）
 */
export function computeRegionsFromPolygons(
  polygons: Point[][],
  worldBounds: { xMin: number; xMax: number; yMin: number; yMax: number }
): Point[][][] {
  if (polygons.length === 0) {
    // 无图形时，整个边界为一个区域（无洞）
    const boundingBox: Point[] = [
      { x: worldBounds.xMin, y: worldBounds.yMin },
      { x: worldBounds.xMax, y: worldBounds.yMin },
      { x: worldBounds.xMax, y: worldBounds.yMax },
      { x: worldBounds.xMin, y: worldBounds.yMax },
      { x: worldBounds.xMin, y: worldBounds.yMin },
    ];
    return [[boundingBox]];
  }

  // 转换所有多边形到 clipping 格式
  const clippingPolys = polygons.map(poly => toClippingFormat(poly));
  
  // 计算所有图形的并集
  const union = polygonClipping.union(...clippingPolys);
  
  // 世界边界矩形（逆时针）
  const boundingBox: number[][][] = [[
    [worldBounds.xMin, worldBounds.yMin],
    [worldBounds.xMax, worldBounds.yMin],
    [worldBounds.xMax, worldBounds.yMax],
    [worldBounds.xMin, worldBounds.yMax],
    [worldBounds.xMin, worldBounds.yMin],
  ]];
  
  // 边界矩形减去并集 = 背景区域（可能多个，带孔）
  const difference = polygonClipping.difference(boundingBox, union);
  
  // 转换回 Point[][][] 格式
  return difference.map(poly => poly.map(ring => ring.map(p => ({ x: p[0], y: p[1] }))));
}

/**
 * 射线法判断点是否在多边形内（含边界）
 */
export function isPointInPolygon(point: Point, polygon: Point[]): boolean {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const xi = polygon[i].x, yi = polygon[i].y;
    const xj = polygon[j].x, yj = polygon[j].y;
    const intersect = ((yi > point.y) !== (yj > point.y)) &&
      (point.x < (xj - xi) * (point.y - yi) / (yj - yi) + xi);
    if (intersect) inside = !inside;
  }
  return inside;
}

/**
 * 根据点击点找到所在区域（背景区域，排除内洞）
 * @param point 世界坐标点
 * @param regions 区域列表，每个区域是多环多边形 Point[][]
 * @returns 命中的区域多边形（多环），未命中返回 null
 */
export function findRegionByPoint(
  point: Point,
  regions: Point[][][]
): Point[][] | null {
  for (const region of regions) {
    if (region.length === 0) continue;
    const outerRing = region[0];
    if (!isPointInPolygon(point, outerRing)) continue;
    // 检查是否落在内洞中
    let insideHole = false;
    for (let i = 1; i < region.length; i++) {
      if (isPointInPolygon(point, region[i])) {
        insideHole = true;
        break;
      }
    }
    if (!insideHole) return region;
  }
  return null;
}

八、贝塞尔曲线与其他图形组合区域识别的保证
贝塞尔曲线闭合判断：用户绘制二次贝塞尔曲线时，通常首尾点自动闭合（双击或自动闭合逻辑）。在 computeApproximatePolygon 中，对于 quadratic 类型，我们默认它是闭合的（由交互保证）。如果实际未闭合，近似多边形将是一条开放折线，此时不应参与区域检测。可以在 isClosedShape 中增加距离阈值判断，若首尾距离大于 1e-6 则视为开放，返回空数组。

采样精度：quadratic 采样 40 个点，足够平滑，且不会自交（二次贝塞尔曲线无自交）。

图形组合：polygon-clipping 能够正确处理多个多边形的并集、差集，无论这些多边形是矩形、圆形（多边形近似）、三角形、闭合贝塞尔曲线、闭合画笔路径。最终得到的背景区域（边界矩形减去并集）能够正确反映由这些图形围出来的独立区域（包括带孔区域）。

开放曲线：线段、开放画笔、开放贝塞尔曲线不产生近似多边形，因此不会影响区域划分。这符合预期，因为它们不封闭，不定义区域边界。

图层独立：每个图层的区域缓存独立，注释只属于其所在图层。

九、迁移与兼容
原有的 annotations 数组可以保留并逐步废弃，或者直接删除。为了平滑过渡，可以在加载旧数据时，将旧注释转换为点注释（因为旧注释本质上是点注释或基于图形的注释，但为了简化，统一转为点注释，位置取几何中心）。

需要清理原有 Shape 上的 annotation 字段，不再使用。

十、总结
通过上述重构：

点注释与区域注释完全分离，职责清晰。

区域注释基于精确的多边形布尔运算，能够正确处理由矩形、圆形、三角形、闭合贝塞尔曲线、闭合画笔等任意组合形成的复杂区域（包括带孔区域）。

每个图形在创建/更新时自动生成近似多边形，并触发区域缓存刷新，确保区域检测实时准确。

性能良好，区域缓存按图层存储，只在图形变化时重新计算。

注意：需要确保 polygon-clipping 库的类型定义可用（如安装 @types/polygon-clipping）。此外，对于超大点数（如画笔数千点）的近似多边形，布尔运算可能稍慢，但一般场景下可接受。如有性能问题，可对画笔路径进行简化（Ramer-Douglas-Peucker 算法）后再存入 approximatePolygon。
